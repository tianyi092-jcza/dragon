async function api(path, value, options = {}) {
  const mark = sessionEpoch; const headers = {}; if (value !== undefined) { headers['Content-Type'] = 'application/json'; headers['Idempotency-Key'] = options.key ?? crypto.randomUUID(); if (options.revision !== undefined) headers['If-Match'] = '"' + options.revision + '"'; if (current) headers['X-CSRF-Token'] = current.csrf; }
  const response = await fetch(path, { method: value === undefined ? 'GET' : 'POST', credentials: 'same-origin', cache: 'no-store', headers, body: value === undefined ? undefined : JSON.stringify(value) });
  if (response.ok && (options.binary || options.libraryBinary)) {
    const limit = options.libraryBinary ? 16 * 1024 * 1024 : 4 * 1024 * 1024;
    if (!Number.isSafeInteger(options.maxBytes) || options.maxBytes < 1 || options.maxBytes > limit || response.headers.get('Content-Type') !== 'application/octet-stream' || !response.body) throw new Error('產物回應格式無效。');
    const parts = [], reader = response.body.getReader(); let length = 0;
    try { while (true) { const part = await reader.read(); if (part.done) break; length += part.value.length; if (length > options.maxBytes) { await reader.cancel(); throw new Error('產物超出固定描述長度。'); } parts.push(part.value); } } finally { reader.releaseLock(); }
    const bytes = new Uint8Array(length); let offset = 0; for (const part of parts) { bytes.set(part, offset); offset += part.length; }
    if (options.libraryBinary) return { bytes, sha256: response.headers.get('X-Content-SHA256'), draftRevision: response.headers.get('X-Draft-Revision'), sourceDigest: response.headers.get('X-Source-SHA256'), catalogRoot: response.headers.get('X-Library-Root'), admission: response.headers.get('X-Content-Admission'), disposition: response.headers.get('Content-Disposition') };
    return { bytes, sha256: response.headers.get('X-Content-SHA256') };
  }
  let data; if (response.ok && options.stagePlan) {
    if (!/^application\/json(?:;|$)/i.test(response.headers.get('Content-Type') ?? '') || !response.body) throw new Error('結構回應格式無效。');
    const reader = response.body.getReader(), parts = []; let length = 0;
    try { while (true) { const part = await reader.read(); if (part.done) break; length += part.value.length; if (length > 1024 * 1024) { await reader.cancel(); throw new Error('結構回應超過1MiB展示預算。'); } parts.push(part.value); } } finally { reader.releaseLock(); }
    const bytes = new Uint8Array(length); let offset = 0; for (const part of parts) { bytes.set(part, offset); offset += part.length; }
    try { data = JSON.parse(new TextDecoder('utf-8', { fatal: true }).decode(bytes)); } catch { throw new Error('結構回應無效。'); }
    return data;
  }
  try { data = await response.json(); } catch { throw new Error('服務回應無效。'); }
  if (!response.ok) { if (response.status === 401 && mark === sessionEpoch) { current = null; sessionEpoch++; render(); } const error = new Error(errors[data.error] ?? '操作被拒絕：' + data.error); error.status = response.status; error.code = data.error; if (data.detail !== undefined) error.detail = data.detail; throw error; }
  return data;
}

function permittedView(next) {
  if (!current || current.user.mustChangePassword) return null;
  if (['profile', 'password'].includes(next)) return next;
  const item = views.find(([id]) => id === next);
  return item && (!item[2] || current.user.role === 'admin') ? next : 'draft-editor';
}

let shownView = null, autoLoading = false;
function show(next) {
  const target = permittedView(next), entered = target !== shownView;
  view = target; shownView = target;
  shell.sync({ current, view, busy });
  // 資源頁是純查詢頁，不顯示上一操作的殘留提示
  if (view === 'catalog-editor' || view === 'copy') status('');
  if (entered && current && !busy && !autoLoading) {
    const load = autoLoad[view];
    if (load) { autoLoading = true; Promise.resolve(load()).catch(() => {}).finally(() => { autoLoading = false; }); }
  }
}

  if (busy || !current || current.user.mustChangePassword || next === view || !canLeave()) return;

function render() {

  const owner = current?.user.id ?? null;
  if (owner !== displayedOwner) {
    displayedOwner = owner;

async function run(action) {
  if (busy) return; busy = true; shell.sync({ current, view, busy }); const controls = [...document.querySelectorAll('#login input,#profile input,#password input')]; controls.forEach(node => { node.disabled = true; });
  // 動作若能說清楚自己完成了什麼（例如「複製完成。」「檢驗完成。」），就在上方提示那句話；
  // 才退回仍有的中性提示，不再讓所有操作都顯示成「操作完成」。
  try { const result = await action(); status(typeof result === 'string' && result ? result : '操作完成。'); return result; } catch (error) { status(error.message); } finally { busy = false; controls.forEach(node => { node.disabled = false; }); draftUI.render(); stageUI?.render(); libraryUI?.render(); catalogUI?.render(); managementUI?.render(); deletionUI?.render(); publishUI?.render(); reviewUI?.render(); stageAssetUI?.render(); strongholdUI?.render(); generalsUI?.render(); mapviewUI?.render(); copyUI.render(); fullCopyUI.render();

    if (draftRefreshPending) { draftRefreshPending = false; void draftUI.actions.refresh(); }
    show(view); }
}
