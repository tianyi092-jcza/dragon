// Selection coordination carries IDs only, never receipts, revisions or game data.
// The caller supplies trusted UI participants; backend ownership remains separate.
export function createAuthoringContext({ participants, initialGame = () => null, beforeChange = () => true }) {
  let selected = null, generation = 0;
  const current = () => selected ?? initialGame();
  return Object.freeze({ current, reset() { selected = null; generation++; }, select(gameId, source) {
    const members = participants(), mark = generation;
    if (typeof gameId !== 'string' || !gameId || !members[source]) return false;
    if (gameId === current()) return true;
    const canAdopt = () => {
      for (const member of Object.values(members)) {
        if (generation !== mark || !member.canAdoptGame(gameId) || generation !== mark) return false;
      }
      return true;
    };
    if (!canAdopt()) return false;
    // Stop at each callback boundary, not only after all consent dialogs.
    for (const [key, member] of Object.entries(members)) {
      if (key !== source && (!member.canLeave() || generation !== mark)) return false;
    }
    if (generation !== mark || !beforeChange() || generation !== mark || !canAdopt()) return false;
    selected = gameId; const committed = ++generation;
    for (const [key, member] of Object.entries(members)) {
      if (key !== source) member.adoptGame(gameId);
      if (generation !== committed) return false;
    }
    return true;
  } });
}

    if (!['material-capture', 'material-rename', 'material-remove', 'decoration-translate', 'decoration-reorder', 'decoration-place', 'decoration-split', 'decoration-remove'].includes(op)) throw new Error('未知素材操作。');

      edit = { op, materialId: selection, variantRef, x, y, waterClass }; label = selection;
    } else if (capturing || moving || reordering || removing) {
      const available = new Set(materials.instances.map(row => row.instanceId));
      if (!Array.isArray(selection) || !selection.length || new Set(selection).size !== selection.length || Array.from(selection).some(id => !available.has(id))) throw new Error('請明確選擇不重複的完整裝飾實例。');

      if (moving) {
        const { dx, dy } = displacement ?? {};
        if (!Number.isSafeInteger(dx) || !Number.isSafeInteger(dy) || Object.is(dx, -0) || Object.is(dy, -0) || dx === 0 && dy === 0) throw new Error('請明確輸入非零整數格位移，不能留空或使用小數。');
        edit.dx = dx; edit.dy = dy;
      }
      if (reordering) {
        const direction = displacement?.direction;
        if (direction !== 1 && direction !== -1) throw new Error('請明確選擇裝飾上移或下移。');
        edit.direction = direction;
      }
    } else {
      const selected = materials.rows.find(row => row.materialId === selection);
      if (!selected?.authorEditable || op === 'material-remove' && selected.references.length) throw new Error('原配方唯讀；有引用的素材不能移除。');
      edit = { op, materialId: selected.materialId }; label = (selected.name ?? '') + ' [' + selected.materialId + ']';
    }

    else if (placing) description = '放置變體 [' + edit.variantRef + '] 於 (' + edit.x + ',' + edit.y + ')，水域類別：' + (edit.waterClass ?? '不指定水域組合') + '；服務端核驗完整占格，不裁切、不自動判定地形；草稿保存不代表可以試玩。';
    else if (moving) description = '整批平移 X ' + edit.dx + '／Y ' + edit.dy + ' 格；可能露出未知底層，草稿保存不代表可以試玩。服務端核驗全部占格範圍，不裁切。';
    else if (reordering) description = (edit.direction === 1 ? '上移（較後繪製）' : '下移（較先繪製）') + '：每個連續選中塊越過一個相鄰未選實例，按整個裝飾層而非篩選頁計算；重疊水域摘要可能改變，不改城市或道路。';
    else if (op === 'material-remove') description = '只移除此修訂的無引用作者素材；不刪除舊修訂或共享資源。';
    else description = '保存素材「' + edit.name + '」；不放置實例、不改地圖。';
    if (!confirm(label + '：' + description + '不保存其它表單輸入。')) return;
    if (!current()) return;
    const body = { metadata: { ...source.metadata }, resourceEdits: [], mapEdits: [edit] };
    if (new TextEncoder().encode(JSON.stringify(body)).byteLength > 16384) throw new Error('素材請求超過16KiB。');
    attempt = { gameId: source.gameId, revision: source.draftRevision, key: crypto.randomUUID(), body, state: 'unknown' };
    if (!persist()) { attempt = null; throw new Error('素材保存未發送：無法保留原操作鍵。'); }
    return sendAttempt('正在保存素材或裝飾操作；回應不明時先查詢原鍵。');
  }

  function checkedWaterMembers(rows, groups) {
    if (rows === undefined || rows === null) return null;
    if (!groups || !Array.isArray(rows) || rows.length !== groups.length) throw new Error('水域成員清單不完整。');
    const members = new Set(), cells = new Set();
    return Array.from(rows, (row, index) => {
      if (!row || Array.isArray(row) || Object.keys(row).sort().join(',') !== 'baseCells,groupId,memberIds' || row.groupId !== groups[index].groupId || !Array.isArray(row.memberIds) || !Array.isArray(row.baseCells) || row.memberIds.length !== groups[index].memberCount || row.baseCells.length !== groups[index].baseCellCount) throw new Error('水域成員身份或計數不符。');
      const memberIds = Array.from(row.memberIds), baseCells = Array.from(row.baseCells);
      for (const id of memberIds) { if (typeof id !== 'string' || !id.length || id.length > 256 || members.has(id)) throw new Error('水域成員身份無效或重複。'); members.add(id); }
      for (const cell of baseCells) { if (!Number.isSafeInteger(cell) || Object.is(cell, -0) || cell < 0 || cell >= 384 * 256 || cells.has(cell)) throw new Error('水域基礎格無效或重複。'); cells.add(cell); }
      return { groupId: row.groupId, memberIds, baseCells };
    });
  }

  }

  async function submitDefinition(remove = false) {
    if (!definitionReady() || attempt || badJournal || !permitted()) throw new Error('請先載入可編輯的武將庫修訂並確認原保存。');
    const source = loaded, form = definitionForm, valid = ticket();
    let edit;
    if (remove) {
      const row = definitions.find(item => item.generalId === form.generalId);
      if (!row) throw new Error('請先選擇已有的人物定義。');
      const references = definitionReferenceRows(row.generalId);
      if (references.length) throw new Error('人物仍有章節配置，包含未加入的配置也須先移除：' + references.map(reference => chapters.find(chapter => chapter.chapterId === reference.chapterId).name + ' [' + reference.chapterId + ']').join('、'));
      if (!confirm('刪除未加入章節的人物定義「' + row.name + '」？本表未保存輸入會捨棄；僅修改草稿，原章槽位不受影響。')) return;
      edit = { op: 'delete', generalId: row.generalId };
    } else {
      const values = { ...definitionText(form.name, form.hao), portraitRef: checkedPortrait(form.portraitRef) };
      edit = form.generalId === null ? { op: 'create', values } : { op: 'update', generalId: form.generalId, values };
      if (form.generalId !== null) {
        const saved = definitions.find(row => row.generalId === form.generalId);
        const changed = ['name', 'hao', 'portraitRef'].filter(key => JSON.stringify(saved[key]) !== JSON.stringify(values[key]));
        const impacts = definitionReferenceRows(form.generalId).map(reference => ({ chapterId: reference.chapterId, fields: reference.inheritedFields.filter(key => changed.includes(key)) })).filter(row => row.fields.length);
        if (impacts.length) {
          const labels = { name: '姓名', hao: '字', portraitRef: '頭像' };
          const details = impacts.map(impact => chapters.find(chapter => chapter.chapterId === impact.chapterId).name + ' [' + impact.chapterId + ']：' + impact.fields.map(key => labels[key]).join('、')).join('\n');
          if (!confirm('修改基礎人物會同步以下章節未覆蓋的欄位（含未加入配置）：\n' + details + '\n確定保存？明確覆蓋不變，只修改草稿。')) return;
          edit.confirmedImpacts = impacts;
        }
      }
    }
    if (!valid() || !permitted() || loaded !== source || definitionForm !== form || attempt || !definitionReady()) return;
    const body = { metadata: { ...source.metadata }, resourceEdits: [], generalEdits: [edit] };
    if (new TextEncoder().encode(JSON.stringify(body)).byteLength > 16384) throw new Error('人物保存請求超過16KiB；請減少輸入，不會截斷文字。');
    attempt = { gameId: source.gameId, revision: source.draftRevision, key: crypto.randomUUID(), body, state: 'unknown' };
    if (!persist()) { attempt = null; throw new Error('人物保存未發送：無法保留原操作鍵。'); }
    return sendAttempt('正在保存人物定義；回應不明時只能查詢原鍵，不另建人物。');
  }

  async function saveNativeCityInputs(chapterId, idx) {

    const patch = nativeCityInputs.get(JSON.stringify([chapterId, idx]));
    if (!row || !patch) return;

      const value = text.trim() === '' ? NaN : Number(text), cap = row[key === 'troops' ? 'troopsCap' : 'maxProd'];

      if (value !== row[key]) values[key] = value;
    }
    if (!Object.keys(values).length) { nativeCityInputs.delete(JSON.stringify([chapterId, idx])); message = '數值與已讀資料相同；沒有保存。'; render(); return; }

  async function saveChapterStart(chapterId) {

      const value = Number(text.trim() === '' ? NaN : text), max = field === 'year' ? 9999 : field === 'month' ? 12 : 31;
      if (!Number.isSafeInteger(value) || Object.is(value, -0) || value < 1 || value > max) throw new Error((field === 'year' ? '年' : field === 'month' ? '月' : '日') + '須為 1 至 ' + max + ' 的整數。');

  async function assignDiplomat(chapterId, factionIdx, generalSlot) {

    if (!faction) throw new Error('請選擇本章已宣告的勢力槽。');
    if (generalSlot === null) {
      if (faction.diplomatSlot === null) { message = '此勢力沒有外交官；沒有保存。'; render(); return; }
      await submitChapterField({ op: 'set-diplomat', chapterId, factionIdx, generalIdx: null }, '清空章節「' + chapter.name + '」勢力槽 ' + factionIdx + ' 的外交官職務（原槽 ' + faction.diplomatSlot + '）');

    }
    const general = roles.generals.find(row => row.slot === generalSlot);
    if (!general) throw new Error('請選擇本章已讀取的武將槽。');
    if (general.reserved || generalSlot === 127) throw new Error('外交官必須是本章的普通武將槽。');
    if ((general.factionSlot ?? null) !== factionIdx) throw new Error('外交官必須先屬於該勢力；請先在武將面設定所屬勢力。');
    const other = heldPosts(roles, generalSlot).filter(post => !(post === postNames.diplomat && faction.diplomatSlot === generalSlot));
    if (other.length) throw new Error('此武將已擔任' + other.join('、') + '；請先清空該職務。');
    if (faction.diplomatSlot === generalSlot) { message = '此武將已是該勢力外交官；沒有保存。'; render(); return; }
    await submitChapterField({ op: 'set-diplomat', chapterId, factionIdx, generalIdx: generalSlot }, '將章節「' + chapter.name + '」勢力槽 ' + factionIdx + ' 的外交官設為武將 #' + generalSlot + '「' + general.name + '」（原槽 ' + (faction.diplomatSlot ?? '無') + '；不改所屬與其它職務）');
  }
  async function submitChapters(edits) {
    if (!loaded?.verifiedRead || !permitted() || badJournal) throw new Error('請先載入自己的精確草稿。');
    if (attempt) throw new Error('先查詢原保存，不另起鍵。');
    attempt = { gameId: loaded.gameId, revision: loaded.draftRevision, key: crypto.randomUUID(), body: { metadata: normalized(), resourceEdits: normalizedResources(), chapterEdits: edits }, state: 'unknown' };
    if (!persist()) { attempt = null; throw new Error('章節保存未發送：無法保留原操作鍵。'); }
    return sendAttempt('正在保存章節修改；如失去回應，保留原輸入與操作鍵並查詢。');
  }

  // 只在送出前組一次命令：名稱沿用章節改名同一套 trim＋NFC，年代只接受整數。
  function chapterCreateRequest() {
    const name = chapterCreateInputs.name.trim().normalize('NFC');
    if (!name) throw new Error('請輸入章節名稱。');
    const start = { era: chapterCreateInputs.era }, labels = { year: '年', month: '月', day: '日' }, limits = { year: 9999, month: 12, day: 31 };
    for (const key of ['year', 'month', 'day']) {
      const text = chapterCreateInputs[key].trim();
      if (!/^[0-9]{1,5}$/.test(text)) throw new Error(`請以整數輸入開始${labels[key]}。`);
      start[key] = Number(text);
      if (start[key] < 1 || start[key] > limits[key]) throw new Error(`開始${labels[key]}須在 1～${limits[key]}。`);
    }
    return { op: 'add-blank', name, start };
  }
  function createChapter() {
    return action(async () => {
      if (!chapterCreateReady()) throw new Error('請先載入含基礎據點資料的精確修訂，才能新增章節。');
      const edit = chapterCreateRequest();
      if (!await chapterAction(edit)) return;
      resetChapterCreate();
      message = '已新增章節「' + edit.name + '」；據點與武將槽為固定源同槽值、所屬勢力為空，須自行設定勢力後才可試運行。';
    });
  }
  // 舊呼叫路徑（僅 op）保留給既有歷史與相容測試；實際介面一律走 createChapter。
  function chapter(op, chapterId) {
    return action(async () => {
      if (op === 'add-blank') return chapterAction({ op });
      const selected = chapters.find(item => item.chapterId === chapterId);
      if (!selected || !['copy', 'rename', 'delete'].includes(op)) throw new Error('請選擇已載入的章節。');
      const edit = { op, chapterId }, valid = ticket();
      if (op === 'rename') { edit.name = prompt('章節新名稱', selected.name); if (edit.name === null) return; }
      if (op === 'delete' && !confirm('刪除章節「' + selected.name + '」？僅影響草稿，不改變已發布版本；草稿可暫無章節，但屆時禁止發布。')) return;
      if (!valid() || !permitted()) return;
      return chapterAction(edit);
    });
  }
