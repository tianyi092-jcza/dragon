// 資源瀏覽器（Web product decision 2026-10-31 修訂於 2026-11）。
// 釘死圖書館角色只可查看與使用，不可上傳、替換或刪除；
// 唯「據點圖片」分類可由管理員新增／替換／刪除（使用者批准 2026-11），
// 上傳圖寬統一壓為 192（高度等比、最近鄰）、僅 PNG、單張 400K。
// 清單本身 NEVER 是執行／發布許可。
const digest = value => typeof value === 'string' && /^[a-f0-9]{64}$/.test(value);
const CATEGORIES = [['portrait', '頭像'], ['image', '圖像／圖塊'], ['city-picture', '據點圖片'], ['audio', '音樂與音效'], ['font', '字體'], ['data', '其他']];
const CATEGORY_IDS = new Set(CATEGORIES.map(([id]) => id));
const CITY_PICTURE_URL_RE = /^\/city-picture\/([0-9]|[1-9][0-9]|1[0-9]{2}|2[0-4][0-9]|25[0-5])\.png(\?sha=[0-9a-f]{64})?$/;
function accept(data) {
  if (!data || data.mode !== 'BUILTIN_READONLY' || !Array.isArray(data.resources)) throw new Error('資源清單回應無效。');
  if (data.resources.length > 4096) throw new Error('資源清單過大。');
  const out = [], seen = new Set();
  for (const row of data.resources) {
    if (!row || typeof row.path !== 'string' || !row.path || row.path.length > 512 || seen.has(row.path)) throw new Error('資源條目無效。');
    if (!Number.isSafeInteger(row.byteLength) || row.byteLength < 1 || !digest(row.sha256) || !CATEGORY_IDS.has(row.category)) throw new Error('資源條目無效。');
    let url = null, width = null, height = null;
    if (row.category === 'city-picture') {
      // 自管圖走自管位址；kyo 凍結圖走館藏位址（須與本行 path 一致，不接受外來位址）。
      const assetUrl = '/api/resources/asset?path=' + encodeURIComponent(row.path);
      if (row.url === assetUrl) url = row.url;
      else if (typeof row.url === 'string' && CITY_PICTURE_URL_RE.test(row.url)) url = row.url;
      else throw new Error('資源條目無效。');
      if (!Number.isSafeInteger(row.width) || !Number.isSafeInteger(row.height) || row.width < 1 || row.height < 1 || row.width > 8192 || row.height > 8192) throw new Error('資源條目無效。');
      width = row.width; height = row.height;
    } else if (row.url !== undefined) throw new Error('資源條目無效。');
    seen.add(row.path);
    out.push(Object.freeze({ path: row.path, byteLength: row.byteLength, category: row.category, label: row.path.replace(/^.*\//, ''), url, width, height }));
  }
  return out;
}
export function installLibraryRegistration({ api, authority, work, isBusy, publish = () => {} }) {
  let actor = null, generation = 0, rows = [], loaded = false, autoInFlight = false, autoAttempts = 0, filter = '', category = 'portrait', message = '';
  const permitted = () => actor && authority() === actor && !actor.user.mustChangePassword;
  const ticket = () => ({ actor, generation });
  const valid = item => permitted() && item.actor === actor && item.generation === generation;
  function visible() {
    const needle = filter.trim().toLowerCase();
    return rows.filter(row => row.category === category && (!needle || row.path.toLowerCase().includes(needle)));
  }
  function render() {
    const allowed = permitted(), counts = Object.create(null);
    for (const row of rows) counts[row.category] = (counts[row.category] || 0) + 1;
    publish({
      categories: CATEGORIES.map(([id, label]) => ({ id, label, count: counts[id] || 0 })),
      category: allowed ? category : 'portrait',
      canManage: allowed && actor.user.role === 'admin',
      filter: allowed ? filter : '',
      rows: allowed ? visible().map(row => Object.freeze({ ...row, url: row.url ?? '/api/resources/asset?path=' + encodeURIComponent(row.path) })) : [],
      total: allowed ? rows.length : 0,
      loaded: allowed && loaded,
      message: allowed ? message : '',
      loading: Boolean(isBusy()),
    });
    // The list is built into the game, so it loads itself once a session is
    // ready. It bypasses the busy/status wrapper so a background fetch never
    // reports "操作完成" or blocks the user, and it retries a few times instead of
    // latching on a stale ticket.
    if (permitted() && !loaded && !autoInFlight && autoAttempts < 3) {
      autoInFlight = true; autoAttempts += 1;
      queueMicrotask(() => { void autoLoad(); });
    }
  }
  async function autoLoad() {
    try {
      const data = await api('/api/resources');
      if (!permitted()) return;
      rows = accept(data); loaded = true; message = '';
    } catch (error) {
      if (permitted()) message = '讀取資源清單失敗：' + (error && error.message ? error.message : error) + '（請確認連線後重試；仍失敗就把這句話告訴管理員）';
    } finally {
      autoInFlight = false; render();
    }
  }
  async function load() {
    if (!permitted()) return;
    const t = ticket();
    message = '正在讀取內置資源清單…'; render();
    const data = await api('/api/resources');
    if (!valid(t)) return;
    rows = accept(data); loaded = true; autoAttempts = 0;
    message = '共 ' + rows.length + ' 項；據點圖片可由管理員新增／替換／刪除，其餘只可查看與使用。';
  }
  async function refresh() {
    const data = await api('/api/resources');
    rows = accept(data); loaded = true;
  }
  // dataUrl 形如 data:image/png;base64,…；服務端再做魔數＋上限＋解碼複核。
  function base64Of(dataUrl) {
    if (typeof dataUrl !== 'string' || !dataUrl.startsWith('data:image/png;base64,')) throw new Error('只接受 PNG 圖片。');
    const raw = dataUrl.slice('data:image/png;base64,'.length);
    if (!raw.length || raw.length > 560 * 1024) throw new Error('圖片過大（單張上限 400K）。');
    return raw;
  }
  async function mutate(action, payload) {
    if (!permitted() || actor.user.role !== 'admin') throw new Error('只有管理員可以管理據點圖片。');
    const t = ticket();
    message = '據點圖片處理中…'; render();
    try {
      const body = { action, ...payload };
      if (body.png !== undefined) body.png = base64Of(body.png);
      await api('/api/city-pictures', body);
      if (!valid(t)) return;
      await refresh();
      if (!valid(t)) return;
      message = action === 'delete' ? '已刪除該據點圖片（引用它的據點會顯示為空）。' : '已保存；寬已統一壓為 192（高度等比）。';
    } catch (error) {
      if (!valid(t)) return;
      message = '據點圖片保存失敗：' + (error && error.message ? error.message : error);
    }
    render();
  }
  const actions = Object.freeze({
    load: () => work(load),
    selectCategory(id) { if (!permitted() || !CATEGORY_IDS.has(id)) return; category = id; render(); },
    setFilter(value) { if (!permitted() || typeof value !== 'string' || value.length > 128) return; filter = value; render(); },
    uploadCityPicture(dataUrl) { return work(() => mutate('create', { png: dataUrl })); },
    replaceCityPicture(view, dataUrl) {
      if (!Number.isInteger(view)) throw new Error('圖片序號無效。');
      return work(() => mutate('replace', { view, png: dataUrl }));
    },
    deleteCityPicture(view) {
      if (!Number.isInteger(view)) throw new Error('圖片序號無效。');
      return work(() => mutate('delete', { view }));
    },
  });
  render();
  return Object.freeze({ render, actions,
    setIdentity(next) {
      if (actor === next) return render();
      generation++; actor = next; rows = []; loaded = false; autoInFlight = false; autoAttempts = 0; filter = ''; category = 'portrait'; message = ''; render();
    },
    pagehide() { generation++; render(); },
    canLeave() { return true; },
  });
}
