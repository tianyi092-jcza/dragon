// Built-in read-only resource browser (Web product decision 2026-10-31).
// Lists the pinned library roles so authors can look at what ships with the
// game. There is no registration, upload, replacement or deletion here, and
// the list is NEVER an execution/publication permission.
const digest = value => typeof value === 'string' && /^[a-f0-9]{64}$/.test(value);
const CATEGORIES = [['portrait', '頭像'], ['image', '圖像／圖塊'], ['audio', '音樂與音效'], ['font', '字體'], ['data', '其他']];
const CATEGORY_IDS = new Set(CATEGORIES.map(([id]) => id));
function accept(data) {
  if (!data || data.mode !== 'BUILTIN_READONLY' || !Array.isArray(data.resources)) throw new Error('資源清單回應無效。');
  if (data.resources.length > 4096) throw new Error('資源清單過大。');
  const out = [], seen = new Set();
  for (const row of data.resources) {
    if (!row || typeof row.path !== 'string' || !row.path || row.path.length > 512 || seen.has(row.path)) throw new Error('資源條目無效。');
    if (!Number.isSafeInteger(row.byteLength) || row.byteLength < 1 || !digest(row.sha256) || !CATEGORY_IDS.has(row.category)) throw new Error('資源條目無效。');
    seen.add(row.path);
    out.push(Object.freeze({ path: row.path, byteLength: row.byteLength, category: row.category, label: row.path.replace(/^.*\//, '') }));
  }
  return out;
}
export function installLibraryRegistration({ api, authority, work, isBusy, publish = () => {} }) {
  let actor = null, generation = 0, rows = [], loaded = false, autoInFlight = false, autoAttempts = 0, filter = '', category = 'portrait', message = '';
  const permitted = () => actor && authority() === actor && !actor.user.mustChangePassword;
  const ticket = () => ({ actor, generation });
  const valid = item => permitted() && item.actor === actor && item.generation === generation;
  function visible() {
    const needle = filter.trim().toLowerCase();
    return rows.filter(row => row.category === category && (!needle || row.path.toLowerCase().includes(needle)));
  }
  function render() {
    const allowed = permitted(), counts = Object.create(null);
    for (const row of rows) counts[row.category] = (counts[row.category] || 0) + 1;
    publish({
      categories: CATEGORIES.map(([id, label]) => ({ id, label, count: counts[id] || 0 })),
      category: allowed ? category : 'portrait',
      filter: allowed ? filter : '',
      rows: allowed ? visible().map(row => Object.freeze({ ...row, url: '/api/resources/asset?path=' + encodeURIComponent(row.path) })) : [],
      total: allowed ? rows.length : 0,
      loaded: allowed && loaded,
      message: allowed ? message : '',
      loading: Boolean(isBusy()),
    });
    // The list is built into the game, so it loads itself once a session is
    // ready. It bypasses the busy/status wrapper so a background fetch never
    // reports "操作完成" or blocks the user, and it retries a few times instead of
    // latching on a stale ticket.
    if (permitted() && !loaded && !autoInFlight && autoAttempts < 3) {
      autoInFlight = true; autoAttempts += 1;
      queueMicrotask(() => { void autoLoad(); });
    }
  }
  async function autoLoad() {
    try {
      const data = await api('/api/resources');
      if (!permitted()) return;
      rows = accept(data); loaded = true; message = '';
    } catch (error) {
      if (permitted()) message = '讀取資源清單失敗：' + (error && error.message ? error.message : error);
    } finally {
      autoInFlight = false; render();
    }
  }
  async function load() {
    if (!permitted()) return;
    const t = ticket();
    message = '正在讀取內置資源清單…'; render();
    const data = await api('/api/resources');
    if (!valid(t)) return;
    rows = accept(data); loaded = true; autoAttempts = 0;
    message = '共 ' + rows.length + ' 個內置資源；只可查看與使用，不可上傳、替換或刪除。';
  }
  const actions = Object.freeze({
    load: () => work(load),
    selectCategory(id) { if (!permitted() || !CATEGORY_IDS.has(id)) return; category = id; render(); },
    setFilter(value) { if (!permitted() || typeof value !== 'string' || value.length > 128) return; filter = value; render(); },
  });
  render();
  return Object.freeze({ render, actions,
    setIdentity(next) {
      if (actor === next) return render();
      generation++; actor = next; rows = []; loaded = false; autoInFlight = false; autoAttempts = 0; filter = ''; category = 'portrait'; message = ''; render();
    },
    pagehide() { generation++; render(); },
    canLeave() { return true; },
  });
}
