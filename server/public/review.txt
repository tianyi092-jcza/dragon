// B3 上架審核 controller（Web 產品決定 2026-11）：管理員審核 copy 单槽待审队列。
// 清单走 GET /api/admin/copy/submissions（提交过即保留，只有 submitted 可审核）；拒绝走
// POST /api/games/:id/copy/reject（槽回 empty）；同意走 /copy/approve 配发 ordinal 并覆盖 formal，
// 不走旧 listing 链；删除＝旧不可逆删除链（游戏级，与 copy 无关）。
// 本檔只做展示與操作追蹤，服務端仍是唯一權限。三個動作在服務端天然收斂
// （同鍵重複＝原回執；換鍵重複＝GAME_CHANGED／changed:false／GAME_DELETING），
// 因此每次點擊只用一個新鍵、失敗後一律先重新整理清單核對，不做自動重送。
export function installReview({ api, authority, work, isBusy, onPurged = () => {}, publish: emit = () => {} }) {
 const uuid = v => typeof v === 'string' && /^[a-f0-9]{8}-[a-f0-9]{4}-4[a-f0-9]{3}-[89ab][a-f0-9]{3}-[a-f0-9]{12}$/.test(v);
 const revision = v => typeof v === 'string' && /^[1-9][0-9]{0,30}$/.test(v);
 const ordinal = v => typeof v === 'string' && /^1\.(0|[1-9][0-9]{0,30})$/.test(v);
 const iso = v => typeof v === 'string' && /^\d{4}-\d\d-\d\dT\d\d:\d\d:\d\d\.\d{3}Z$/.test(v) && Number.isFinite(Date.parse(v)) && new Date(v).toISOString() === v;
 const fields = (v, keys) => v && typeof v === 'object' && !Array.isArray(v) && Object.keys(v).sort().join(',') === [...keys].sort().join(',');
 const hex = v => typeof v === 'string' && /^[a-f0-9]{64}$/.test(v);
 const name = v => typeof v === 'string' && v.isWellFormed() && [...v].length >= 1 && [...v].length <= 8;
 const states = { submitted: '待審核', approved: '已通過', empty: '暫無新申請', ready: '已校驗，尚未提交' };
 const errors = {
  ADMIN_REQUIRED: '只有管理員可以審核上架申請。',
  RELEASE_HISTORY_QUERY: '發布歷史查詢格式無效；請重新整理清單。',
  GAME_CHANGED: '清單已被其他人變更（或重複操作）；請重新整理後再決定。',
  REVIEW_LISTED: '此版本已上架；審核只處理待上架的提交。',
  SLOT_NOT_SUBMITTED: '该槽不在已提交状态；请重新整理。',
  REVIEW_QUEUE_CORRUPT: '待审队列损坏；请联系管理员核查。',
  GAME_DELETING: '遊戲已進入刪除流程，不能繼續此操作。',
  BUILTIN_PROTECTED: '內建遊戲不可審核或刪除。',
  DELETE_CONFIRMATION: '刪除確認名稱與目前遊戲名稱不符，請重新整理清單。',
  DELETE_INPUT: '刪除請求格式無效；請重新整理清單。',
  GAME_NOT_FENCED: '尚未建立不可逆刪除柵欄，不能物理刪除；請重走刪除確認。',
  PURGE_CONFIRMATION: '物理刪除確認名稱與遊戲名稱不符，已停止；柵欄仍在，可核对後重試。',
  PURGE_STALE: '游戏在删除确认后又被改动，物理删除已停止；柵欄仍在，请重新整理后重試。',
  PURGE_LISTED: '遊戲仍在玩家端上架中，不能物理刪除；请先下架再重試。',
  PURGE_FORBIDDEN: '只有作者本人或管理員能物理刪除此遊戲。',
  PURGE_INCOMPLETE: '物理刪除复查发现残留列，已整批回滚；请把涉及表名告诉管理员。',
  RECLAIMER_UNCONFIGURED: '檔案清理服務未配置；資料列與刪除入口已保留，配置後可重試。',
  RECLAIM_FAILED: '遊戲檔案清理未完成；資料列與刪除入口已保留，可重試繼續清理。',
  LISTING_RESTRICTED: '此遊戲處於管理限制，不能同意上架。',
  DISTRIBUTION_NOT_CONFIGURED: '服務尚未配置分發狀態；不推定已公開。',
  DISTRIBUTION_GAME_PENDING: '此遊戲仍有待確認分發，不能上架下一版。',
  DISTRIBUTION_STATUS_UNAVAILABLE: '原上架缺少可核對的分發關聯；不推定成功或失敗。',
  MANAGEMENT_RECEIPT_CORRUPT: '原審核回執無效；不推定成功或失敗。',
  IDEMPOTENCY_CONFLICT: '此操作鍵已被其它請求使用；請重新整理後再試。',
  SESSION_INVALID: '會話已失效，請重新登入。',
 };
 let identity = null, generation = 0, rows = [], messageText = '', resultText = '', sending = false;
 // 發布歷史（A 路線日誌）：admin 唯讀，且只在按「發布歷史」時讀一次；
 // 不進 client 的自動載入，免得每次進審核頁都多打一筆請求。
 const releaseStates = { pending: '待確認', confirmed: '已公開' };
 let historyGameId = null, historyRows = [], historyText = '', historyLoading = false;
 const allowed = () => Boolean(identity?.user) && !identity.user.mustChangePassword && identity.user.role === 'admin';
 const message = text => { messageText = text; };
 function current(ctx) { const live = authority(); return allowed() && live?.user?.id === ctx.id && live.user.version === ctx.version && !live.user.mustChangePassword && generation === ctx.generation; }
 function capture() { const ctx = { id: identity?.user.id, version: identity?.user.version, generation }; if (!current(ctx)) throw new Error('請重新核對登入的帳戶。'); return ctx; }
 function rowList(value) {
  if (!value || !Array.isArray(value.queue)) throw new Error('待审队列格式无效。');
  const seen = new Set();
  return value.queue.map(row => {
   if (!row || typeof row !== 'object' || Array.isArray(row) || !uuid(row.gameId) || seen.has(row.gameId)) throw new Error('待审队列栏位无效。');
   if (typeof row.name !== 'string' || !name(row.name) || typeof row.author !== 'string' || [...row.author].length < 1 || [...row.author].length > 32) throw new Error('待审队列栏位无效。');
   if (typeof row.description !== 'string' || [...row.description].length > 2000) throw new Error('待审队列栏位无效。');
   if ((row.submittedAt !== null && !iso(row.submittedAt)) || (row.validatedAt !== null && !iso(row.validatedAt))) throw new Error('待审队列栏位无效。');
   if (row.validation !== null && (!row.validation || typeof row.validation !== 'object' || !['passed', 'failed'].includes(row.validation.state) || !iso(row.validation.checkedAt))) throw new Error('待审队列栏位无效。');
   if ((row.version !== null && !ordinal(row.version)) || !ordinal(row.nextVersion)) throw new Error('待审队列栏位无效。');
   if (!revision(row.rowRevision) || !iso(row.createdAt)) throw new Error('待审队列栏位无效。');
   if (!Object.hasOwn(states, row.state) || typeof row.listed !== 'boolean') throw new Error('審核狀態無效。');
   seen.add(row.gameId); return Object.freeze({ ...row });
  });
 }
 function listingReceipt(value, row) {
  if (!value || typeof value !== 'object' || value.gameId !== row.gameId || value.listed !== true || !revision(value.rowRevision)) throw new Error('上架回執格式無效；請重新整理清單。');
  return value;
 }
 function rejectReceipt(value, row) {
  if (!value || typeof value !== 'object' || value.gameId !== row.gameId || value.state !== 'empty') throw new Error('駁回回執格式無效；請重新整理清單。');
  return value;
 }
 function deleteReceipt(value, row) {
  if (!fields(value, ['createdAt', 'gameId', 'rowRevision', 'state']) || value.gameId !== row.gameId || value.state !== 'fenced' || !iso(value.createdAt) || !revision(value.rowRevision)) throw new Error('刪除回執格式無效；請重新整理清單。');
  // 新栅栏＝行修订+1；已栅栏重试＝回执修订即当前行修订（栅栏只 bump 一次，不可编辑的游戏不会再变）。其余一律拒收。
  if (BigInt(value.rowRevision) !== BigInt(row.rowRevision) + 1n && BigInt(value.rowRevision) !== BigInt(row.rowRevision)) throw new Error('刪除回執格式無效；請重新整理清單。');
  return value;
 }
 // 現行服務先回收檔案再刪庫行；兼容舊服務 pending 回執，但絕不顯示完成。
 const reclaimed = value => value === undefined || value !== null && typeof value === 'object' && !Array.isArray(value) &&
  ['completed', 'pending'].includes(value.state) && Number.isSafeInteger(value.objects) && value.objects >= 0 &&
  Object.keys(value).every(key => ['state', 'code', 'objects', 'privateObjects', 'pointer'].includes(key)) &&
  (value.privateObjects === undefined || Number.isSafeInteger(value.privateObjects) && value.privateObjects >= 0) &&
  (value.state === 'completed' ? ['removed', 'rewritten', 'absent', 'already-absent'].includes(value.pointer) && value.code === undefined : typeof value.code === 'string' && value.code.length >= 1 && value.code.length <= 200 && !/[\x00-\x1f\x7f]/.test(value.code) && value.pointer === undefined);
 const reclaimText = value => !value ? '' : value.state === 'completed'
  ? ' 在架指標與遊戲存儲檔案已回收。'
  : value.code === 'RECLAIMER_UNCONFIGURED' ? ' 存儲回收未設定，檔案可能仍留在儲存。'
  : ' 存儲回收未完成（' + value.code + '），可稍後重試。';
 function purgeReceipt(value, row) {
  if ((!fields(value, ['gameId', 'purgedAt', 'reclamation', 'removed', 'state', 'tables', 'version']) && !fields(value, ['gameId', 'purgedAt', 'reclamation', 'removed', 'retirement', 'state', 'tables', 'version'])) || value.version !== 1 || value.state !== 'purged' || value.gameId !== row.gameId || !iso(value.purgedAt) || !Number.isSafeInteger(value.tables) || !value.removed || typeof value.removed !== 'object' || value.removed.content_games !== 1 || !reclaimed(value.reclamation)) throw new Error('徹底刪除回執格式無效；柵欄已成立，可稍後重試。');
  return value;
 }
 function releaseHistory(value, gameId) {
  if (!fields(value, ['gameId', 'releases']) || value.gameId !== gameId || !Array.isArray(value.releases) || value.releases.length > 1000) throw new Error('發布歷史格式無效。');
  const seen = new Set();
  return value.releases.map(entry => {
   if (!fields(entry, ['digest', 'gameId', 'publishedAt', 'state', 'version']) || entry.gameId !== gameId || !ordinal(entry.version) || !hex(entry.digest) || !Object.hasOwn(releaseStates, entry.state) || seen.has(entry.version)) throw new Error('發布歷史格式無效。');
   // 「待確認」不得帶時間，「已公開」必須帶可核對的時間；兩者不能互相冒充。
   if ((entry.state === 'pending') !== (entry.publishedAt === null) || (entry.publishedAt !== null && !iso(entry.publishedAt))) throw new Error('發布歷史格式無效。');
   seen.add(entry.version); return Object.freeze({ ...entry });
  });
 }
 function render() {
  const visible = current({ id: identity?.user?.id, version: identity?.user?.version, generation }), disabled = !visible || isBusy() || sending;
  const list = visible ? rows.map(row => ({
   gameId: row.gameId, name: row.name, author: row.author, description: row.description,
   version: row.version ?? '—', nextVersion: row.nextVersion, stateKey: row.state, state: states[row.state] + (row.listed ? '／已上架' : '／未上架'), rejected: false,
   createdAt: row.createdAt, stateTime: row.submittedAt ?? row.validatedAt ?? row.createdAt, rowRevision: row.rowRevision,
   approveDisabled: disabled || row.state !== 'submitted', rejectDisabled: disabled || row.state !== 'submitted', deleteDisabled: disabled,
   historyDisabled: disabled, historyOpen: historyGameId === row.gameId && visible,
   approveHint: row.state === 'submitted' ? '同意后版本自动为 ' + row.nextVersion + ' 并覆盖正式版本；失败不耗序号' : '須等待作者再次提交申請才能審核',
   rejectHint: row.state === 'submitted' ? '驳回后槽回空，作者重验可再交' : '目前沒有待審核申請',
   deleteHint: '刪除整個遊戲（含所有章節與已上架版本，含資料庫列），不是刪除單個版本，徹底清除且不可恢復',
  })) : [];
  const history = {
   gameId: visible ? historyGameId : null, loading: visible && historyLoading, message: visible ? historyText : '',
   rows: visible && historyGameId !== null ? historyRows.map(entry => ({
    version: entry.version, state: entry.state, stateText: releaseStates[entry.state], publishedAt: entry.publishedAt ?? '—',
    // 只用前 8 位作摘要，整串留在 title；不把機器串塞進句子裡。
    digest: entry.digest, shortDigest: entry.digest.slice(0, 8),
   })) : [],
  };
  emit({ rows: list, history, message: visible ? messageText : '', result: visible ? resultText : '', locked: disabled, refreshDisabled: !visible || isBusy() || sending });
 }
 async function load(ctx, options) {
  const value = await api('/api/admin/copy/submissions', undefined, options);
  if (!current(ctx)) return;
  rows = rowList(value);
 }
 async function refresh() { const ctx = capture(); message(''); resultText = ''; await load(ctx); render(); return '上架清單已更新。'; }
 // 每個動作：新鍵、CASA 由伺服器判定、無論成敗都重讀清單（狀態只在伺服器成立）。
 async function execute(row, request, verify, done, options) {
  const ctx = capture();
  if (!rows.includes(row)) throw new Error('審核清單已變，請重新整理。');
  sending = true; message(''); resultText = ''; render();
  try {
   const value = await request();
   verify(value, row);
   resultText = typeof done === 'function' ? done(value) : done;
   // 把本次动作的人话结果一并带给外层（页面顶部提示行），别只剩一句「操作完成」。
   return resultText;
  } catch (error) {
   if (current(ctx)) message(errors[error.code] ?? error.message);
   throw error;
  } finally {
   sending = false;
   if (current(ctx)) { try { await load(ctx, options); } catch { /* 清單保持舊值，直到明確重新整理 */ } }
   render();
  }
 }
 const rowFor = (gameId, rowRevision) => {
  const row = rows.find(entry => entry.gameId === gameId && entry.rowRevision === rowRevision);
  if (!row) throw new Error('審核目標已變，請重新整理清單。');
  return row;
 };
 async function approve(gameId, rowRevision) {
  const row = rowFor(gameId, rowRevision);
  if (row.state !== 'submitted') throw new Error('目前沒有待審核申請。');
  return await execute(row, () => api('/api/games/' + row.gameId + '/copy/approve', {}),
    (value, target) => {
      if (!value || typeof value !== 'object' || value.gameId !== target.gameId || value.state !== 'approved' || typeof value.ordinal !== 'string') throw new Error('同意回执无效；请重新整理清单。');
      return value;
    }, value => '已同意上架版本 ' + value.ordinal + '。');
 }
 async function reject(gameId, rowRevision) {
  const row = rowFor(gameId, rowRevision);
  if (row.state !== 'submitted') throw new Error('目前沒有待審核申請。');
  return await execute(row, () => api('/api/games/' + row.gameId + '/copy/reject', {}), rejectReceipt, '已驳回「' + row.name + '」的上架申请；槽已回空，作者重验可再交。');
 }
 async function remove(gameId, rowRevision, onProgress = () => {}) {
  const row = rowFor(gameId, rowRevision);
  let outcome = null;
  await execute(row, async () => {
   // 栅栏已存在（上次 purge 失败回滚）时服务端直接返回既有回执继续 purge，不抛 GAME_DELETING。
   onProgress('fencing');
   const fence = deleteReceipt(await api('/api/games/' + row.gameId + '/deletion', { gameId: row.gameId, expectedRowRevision: row.rowRevision, confirmationName: row.name }, { key: crypto.randomUUID(), feedback: 'progress' }), row);
   onProgress('purging');
   try {
    const purge = purgeReceipt(await api('/api/admin/games/' + row.gameId + '/purge', { confirmationName: row.name, expectedRowRevision: fence.rowRevision }, { key: crypto.randomUUID(), feedback: 'progress' }), row);
    outcome = { ok: true, receipt: purge };
    onProgress('refreshing');
    return { fence, purge };
   }
   catch (error) {
    const tables = Array.isArray(error?.detail?.tables) ? error.detail.tables.filter(t => typeof t === 'string') : [];
    const detail = typeof error?.detail?.reason === 'string' ? '（' + error.detail.reason + '）' : '';
    const text = (errors[error.code] ?? (error.message || error.code || '原因未回報')) + detail;
    outcome = { ok: false, code: error.code, error: text, errorTables: tables };
    return { fence, purgeError: text + (tables.length ? '（涉及：' + tables.join('、') + '）' : '') };
   }
  }, value => value.fence, value => value.purge
   ? (value.purge.reclamation?.state === 'completed' ? '已徹底刪除（不可恢復）：' : '資料列已刪除，但存儲回收尚未確認，不能視為徹底刪除：') + row.name + '；共享程序、共享畫像與玩家本地存檔不受影響。' + reclaimText(value.purge.reclamation)
   : '已建立不可逆刪除柵欄，但物理刪除尚未完成：' + value.purgeError, { feedback: 'progress' });
  // 只有拿到物理刪除回執才算乾淨刪除，同时把回执带给弹窗逐组列状态。
  // 同時告知其他工作頁：這個遊戲是真的被刪了（不是清單遺漏），應丟掉對它的上下文。
  if (outcome?.ok) { try { onPurged(row.gameId); } catch { /* 提示失敗不影響已成立的刪除事實。 */ } }
  return outcome ?? { ok: false, error: '原因未回報', errorTables: [] };
 }
 async function history(gameId) {
  const row = rows.find(entry => entry.gameId === gameId);
  if (!row) throw new Error('審核目標已變，請重新整理清單。');
  // 再按一次就收起：本頁沒有本機待核對的狀態，收起不影響任何伺服器事實。
  if (historyGameId === gameId) { historyGameId = null; historyRows = []; historyText = ''; resultText = '已收起發布歷史。'; render(); return resultText; }
  const ctx = capture();
  historyGameId = gameId; historyRows = []; historyText = ''; historyLoading = true; message(''); resultText = ''; render();
  try {
   const list = releaseHistory(await api('/api/admin/games/' + gameId + '/releases'), gameId);
   if (current(ctx) && historyGameId === gameId) historyRows = list;
   resultText = list.length ? '已讀取「' + row.name + '」的 ' + list.length + ' 筆發布記錄。' : '「' + row.name + '」還沒有發布記錄：發布日誌從這次上線之後才開始記，不會回填舊版本。';
   return resultText;
  } catch (error) {
   // 失敗要看得見、說得清，並保持展開讓使用者可以再試一次。
   if (current(ctx) && historyGameId === gameId) historyText = (errors[error.code] ?? (error.message || error.code || '原因未回報')) + ' 可以再按一次「發布歷史」重試；若一直失敗，請把這句話告訴管理員。';
   throw error;
  } finally {
   historyLoading = false;
   if (current(ctx)) render();
  }
 }
 const actions = Object.freeze({
  history: gameId => work(async () => { const ctx = capture(); try { return await history(gameId); } catch (error) { if (current(ctx)) message(errors[error.code] ?? error.message); throw error; } finally { render(); } }),
  refresh: () => work(async () => { const ctx = capture(); try { return await refresh(); } catch (error) { if (current(ctx)) message(errors[error.code] ?? error.message); throw error; } finally { render(); } }),
  approve: (gameId, rowRevision) => work(async () => { const ctx = capture(); try { return await approve(gameId, rowRevision); } catch (error) { if (current(ctx)) message(errors[error.code] ?? error.message); throw error; } finally { render(); } }),
  reject: (gameId, rowRevision) => work(async () => { const ctx = capture(); try { return await reject(gameId, rowRevision); } catch (error) { if (current(ctx)) message(errors[error.code] ?? error.message); throw error; } finally { render(); } }),
  remove: (gameId, rowRevision, onProgress) => work(async () => { const ctx = capture(); try { return await remove(gameId, rowRevision, onProgress); } catch (error) { if (current(ctx)) message(errors[error.code] ?? error.message); throw error; } finally { render(); } }, { feedback: 'progress' }),
 });
 return Object.freeze({ actions, render,
  setIdentity(value) {
   const next = value?.user ? { user: { id: value.user.id, version: value.user.version, role: value.user.role, mustChangePassword: value.user.mustChangePassword } } : null;
   const changed = JSON.stringify(identity) !== JSON.stringify(next); identity = next;
   if (changed) { generation++; rows = []; message(''); resultText = ''; historyGameId = null; historyRows = []; historyText = ''; historyLoading = false; }
   render();
  },
  // 本頁只有明確伺服器操作，沒有本機待核對的原鍵，因此離開不需要提示。
  canLeave() { return true; } });
}
