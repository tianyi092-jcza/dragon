// Admin management UI only. Session/public version/local journal never grants service authority.
export function installGameManagement({api,authority,work,isBusy}) {
 const q=id=>document.querySelector('#'+id), uuid=v=>typeof v==='string'&&/^[a-f0-9]{8}-[a-f0-9]{4}-4[a-f0-9]{3}-[89ab][a-f0-9]{3}-[a-f0-9]{12}$/.test(v), revision=v=>typeof v==='string'&&v.length<=8192&&/^[1-9][0-9]*$/.test(v);
 let identity=null,generation=0,games=[],pending=null,checked=false,badJournal=false,last=null;
 const allowed=()=>identity?.user.role==='admin'&&!identity.user.mustChangePassword;
 const message=text=>{q('management-message').textContent=text;};
 const journalKey=id=>'editor-management-operation:'+id;
 function tuple(v){return v&&typeof v==='object'&&!Array.isArray(v)&&Object.keys(v).sort().join(',')==='action,actorId,expected,gameId,key,version'&&uuid(v.actorId)&&uuid(v.gameId)&&Number.isSafeInteger(v.version)&&v.version>0&&typeof v.key==='string'&&/^[A-Za-z0-9_-]{16,128}$/.test(v.key)&&['unlist','clear-restriction'].includes(v.action)&&revision(v.expected);}
 function capture(){if(!allowed())throw new Error('僅管理員可管理下架與限制。');return Object.freeze({id:identity.user.id,version:identity.user.version,generation});}
 function current(ctx){const live=authority();return allowed()&&live?.user.id===ctx.id&&live.user.version===ctx.version&&live.user.role==='admin'&&!live.user.mustChangePassword&&generation===ctx.generation;}
 function store(value){if(!identity)throw new Error('請先登入。');try{if(value)sessionStorage.setItem(journalKey(identity.user.id),JSON.stringify(value));else sessionStorage.removeItem(journalKey(identity.user.id));}catch{throw new Error('無法保留管理操作關聯；不發送新請求。');}pending=value;checked=false;badJournal=false;}
 function restore(){pending=null;checked=false;badJournal=false;last=null;if(!allowed())return;try{const raw=sessionStorage.getItem(journalKey(identity.user.id));if(raw!==null){if(raw.length>20000)throw new Error('journal budget');let v;try{v=JSON.parse(raw);}catch{throw new Error('journal JSON');}if(!tuple(v)||v.actorId!==identity.user.id)throw new Error('journal tuple');pending=Object.freeze(v);}}catch{badJournal=true;}}
 function receipt(value,t){if(!value||typeof value!=='object'||Array.isArray(value)||Object.keys(value).sort().join(',')!=='action,eventAt,gameId,listed,restricted,rowRevision'||value.gameId!==t.gameId||value.action!==t.action||value.listed!==false||value.restricted!==(t.action==='unlist')||!revision(value.rowRevision)||typeof value.eventAt!=='string'||!/^\d{4}-\d\d-\d\dT\d\d:\d\d:\d\d\.\d{3}Z$/.test(value.eventAt))throw new Error('管理回執格式無效；保留原鍵並先查詢。');const n=BigInt(value.rowRevision),old=BigInt(t.expected);if(n!==old&&n!==old+1n)throw new Error('管理回執修訂不符；保留原關聯。');return Object.freeze({...value});}
 function list(value){if(!value||!Array.isArray(value.games)||value.games.length>10000)throw new Error('管理清單格式無效。');const seen=new Set();return value.games.map(g=>{if(!g||!uuid(g.gameId)||seen.has(g.gameId)||typeof g.creatorAccount!=='string'||(g.name!==null&&typeof g.name!=='string')||(g.introduction!==null&&typeof g.introduction!=='string')||typeof g.listed!=='boolean'||typeof g.restricted!=='boolean'||!revision(g.rowRevision))throw new Error('管理清單欄位無效。');seen.add(g.gameId);return Object.freeze({gameId:g.gameId,creatorAccount:g.creatorAccount,name:g.name,introduction:g.introduction,listed:g.listed,restricted:g.restricted,rowRevision:g.rowRevision});});}
 function render(){q('management-editor').hidden=!allowed();q('management-games').replaceChildren();for(const g of games){const row=document.createElement('div'),label=document.createElement('p');label.textContent=(g.name??'尚無正式版本')+' — '+g.gameId+'／創作者 '+g.creatorAccount+'／'+(g.listed?'已上架':'下架')+'／'+(g.restricted?'管理限制':'無管理限制')+'／管理修訂 '+g.rowRevision;row.append(label);for(const action of ['unlist','clear-restriction']){const button=document.createElement('button');button.type='button';if(action==='unlist')button.textContent=g.listed?'管理下架並限制':'設管理限制（保持下架）';else button.textContent='解除管理限制（不自動上架）';button.disabled=!allowed()||isBusy()||Boolean(pending)||badJournal||(action==='clear-restriction'&&!g.restricted);button.onclick=()=>work(()=>start(g,action));row.append(button);}q('management-games').append(row);}
 const disabled=!allowed()||isBusy();q('management-load').disabled=disabled;q('management-query').disabled=disabled||!pending;q('management-retry').disabled=disabled||!pending||!checked||pending.version!==identity?.user.version;q('management-forget').disabled=disabled||(!pending&&!badJournal);
 let text='沒有未確認管理操作。';if(badJournal)text='本機管理關聯無法核對；不發送新操作。可明確停止本機追蹤，不取消SQL。';else if(pending)text='未確認原操作 '+pending.key+'／目標 '+pending.gameId+'／'+pending.action+'／原管理修訂 '+pending.expected+(pending.version!==identity?.user.version?'；帳戶版本已變，不重送舊請求。':'')+(checked?'；已明確查詢，仍未取得成功回執。':'；須先明確GET，不自動重試。');q('management-operation').textContent=text;q('management-result').textContent=last?'已確認原操作（歷史事實，不是目前狀態）：'+last.action+'／目標 '+last.gameId+'／回執管理修訂 '+last.rowRevision+'／'+last.eventAt:'';
 }
 async function refresh(){const ctx=capture();const result=await api('/api/admin/games');if(!current(ctx))return;games=list(result);render();}
 async function confirmed(value,t,ctx){const r=receipt(value,t);if(!current(ctx)||pending?.key!==t.key)return;store(null);last=r;render();message('已確認原操作回執；另讀目前管理狀態，不恢復舊狀態。');await refresh();}
 async function send(t,ctx){checked=false;render();const path='/api/games/'+encodeURIComponent(t.gameId)+(t.action==='unlist'?'/listing':'/restriction/clear'),body=t.action==='unlist'?{action:'unlist'}:{};try{const value=await api(path,body,{key:t.key,revision:t.expected});await confirmed(value,t,ctx);}catch(error){if(current(ctx)&&pending?.key===t.key){message(error.message+' 原關聯保留；先查詢，不自動重試或另起鍵。');render();}throw error;}}
 async function start(g,action){const ctx=capture();if(pending||badJournal)throw new Error('請先核對原管理操作，不另起鍵。');const wording=action==='unlist'?'管理下架並限制此遊戲？作者仍可保存修正，不能藉發布或上架繞過限制。':'解除管理限制？遊戲仍保持下架，不自動上架。';if(!confirm(wording+'\n'+g.gameId))return;const t=Object.freeze({actorId:ctx.id,version:ctx.version,gameId:g.gameId,key:crypto.randomUUID(),action,expected:g.rowRevision});store(t);last=null;render();await send(t,ctx);}
 async function query(){const ctx=capture(),t=pending;if(!t)throw new Error('沒有原操作關聯。');checked=false;render();try{const value=await api('/api/admin/games/'+encodeURIComponent(t.gameId)+'/management-operations/'+encodeURIComponent(t.key));await confirmed(value,t,ctx);}catch(error){if(current(ctx)&&pending?.key===t.key){if(error.status===404&&error.code==='MANAGEMENT_OPERATION_NOT_FOUND'){checked=true;message('此GET尚未找到原回執；不證明原請求已取消或不在途。可明確同鍵、同body、原If-Match重試，絕不另鍵或自動重試。');}else message(error.message+' 原關聯保留；不推定成功或取消。');render();}throw error;}}
 async function retry(){const ctx=capture(),t=pending;if(!t||!checked||t.version!==ctx.version)throw new Error('須先明確查詢，且帳戶版本及原完整關聯不能改變。');if(!confirm('明確重試原鍵與原空／unlist body及If-Match？只取得原結果或提交同一操作，不另起鍵。'))return;await send(t,ctx);}
 function forget(){if(!confirm('只停止本機追蹤？不取消在途請求、不回滾已提交SQL，也不表示原操作沒有生效。請重新讀取目前清單。'))return;store(null);last=null;message('已停止本機追蹤；原永久服務鍵不刪除或重用。');render();}
 q('management-load').onclick=()=>work(refresh);q('management-query').onclick=()=>work(query);q('management-retry').onclick=()=>work(retry);q('management-forget').onclick=()=>work(forget);
 return Object.freeze({render,setIdentity(value){const changed=identity?.user.id!==value?.user.id||identity?.user.version!==value?.user.version||identity?.user.role!==value?.user.role||identity?.user.mustChangePassword!==value?.user.mustChangePassword;identity=value;if(changed){generation++;games=[];restore();}render();},canLeave(){return (!pending&&!badJournal)||confirm('管理操作仍未確認。離開只保留本機原關聯，不取消或回滾服務；回來後須明確查詢。');}});
}
