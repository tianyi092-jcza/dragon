// Management/deletion operation tracking. Browser journals never grant service authority.
export function installGameDeletion({api,authority,work,isBusy,publish=()=>{},storage=globalThis.sessionStorage,confirm=text=>globalThis.confirm(text),prompt=text=>globalThis.prompt(text)}) {
 const sessionStorage=storage;
 let messageText='';
 const uuid=v=>typeof v==='string'&&/^[a-f0-9]{8}-[a-f0-9]{4}-4[a-f0-9]{3}-[89ab][a-f0-9]{3}-[a-f0-9]{12}$/.test(v);
 const revision=v=>typeof v==='string'&&/^[1-9][0-9]{0,63}$/.test(v);
 const exact=(v,keys)=>v&&typeof v==='object'&&!Array.isArray(v)&&Object.keys(v).sort().join(',')===keys;
 const iso=v=>typeof v==='string'&&/^\d{4}-\d\d-\d\dT\d\d:\d\d:\d\d\.\d{3}Z$/.test(v)&&Number.isFinite(Date.parse(v))&&new Date(v).toISOString()===v;
 const nameOK=v=>typeof v==='string'&&v.isWellFormed()&&[...v].length<=8;
 let identity=null,generation=0,pending=null,bad=false,checked=false,last=null,distribution=null,serial=0,sending=false,games=[];
 const allowed=()=>identity&&!identity.mustChangePassword&&['author','admin'].includes(identity.role)&&uuid(identity.id)&&Number.isSafeInteger(identity.version)&&identity.version>0;
 const journal=id=>'editor-deletion-operation:'+id;
 const message=text=>{messageText=text;};
 function current(ctx){const user=authority()?.user;return allowed()&&generation===ctx.generation&&user?.id===ctx.id&&user.version===ctx.version&&user.role===ctx.role&&!user.mustChangePassword;}
 function capture(){const ctx={...identity,generation};if(!current(ctx))throw new Error('請先核對目前帳戶。');return ctx;}
 function tuple(v){return exact(v,'actorId,confirmationName,expectedRowRevision,gameId,key,version')&&uuid(v.actorId)&&uuid(v.gameId)&&Number.isSafeInteger(v.version)&&v.version>0&&typeof v.key==='string'&&/^[A-Za-z0-9_-]{16,128}$/.test(v.key)&&revision(v.expectedRowRevision)&&nameOK(v.confirmationName);}
 function store(value){const key=journal(identity.id),text=value?JSON.stringify(value):null;try{if(value)sessionStorage.setItem(key,text);else sessionStorage.removeItem(key);if(sessionStorage.getItem(key)!==text)throw new Error('readback');}catch{bad=true;render();throw new Error('無法保留刪除請求；不發送新操作。');}pending=value;bad=false;checked=false;serial++;}
 function restore(){pending=null;bad=false;checked=false;last=null;distribution=null;serial++;if(!allowed())return;try{const raw=sessionStorage.getItem(journal(identity.id));if(raw!==null){if(raw.length>4096)throw new Error('budget');let v;try{v=JSON.parse(raw);}catch{throw new Error('JSON');}if(!tuple(v)||v.actorId!==identity.id)throw new Error('tuple');pending=Object.freeze(v);}}catch{bad=true;}}
 function receipt(value,t){if(!exact(value,'createdAt,gameId,rowRevision,state')||value.gameId!==t.gameId||value.state!=='fenced'||!revision(value.rowRevision)||BigInt(value.rowRevision)!==BigInt(t.expectedRowRevision)+1n||!iso(value.createdAt))throw new Error('刪除柵欄回執無效；保留原鍵。');if(last&&Object.keys(value).some(k=>value[k]!==last[k]))throw new Error('原刪除回執不一致。');return Object.freeze({...value});}
 function observation(value,t){if(!exact(value,'distribution,gameId,result,version')||value.version!==1||value.gameId!==t.gameId)throw new Error('刪除分發回執無效。');const result=receipt(value.result,t),d=value.distribution;if(!exact(d,'gameId,generation,kind,operationId,recoveryRequired,state,visibleAt')||d.gameId!==t.gameId||d.kind!=='delete-fence'||!uuid(d.operationId)||!Number.isSafeInteger(d.generation)||d.generation<1||typeof d.recoveryRequired!=='boolean'||!['pending','confirmed'].includes(d.state)||(d.state==='pending'?d.visibleAt!==null:!iso(d.visibleAt)))throw new Error('刪除分發狀態無效；不推定清理完成。');if(distribution&&['operationId','generation','kind'].some(k=>d[k]!==distribution[k]))throw new Error('原刪除分發關聯不一致。');return {result,distribution:Object.freeze({...d})};}
 function render(){
  const visible=current({...identity,generation}),disabled=!visible||isBusy()||sending;
  const rows=visible?games.map(g=>({gameId:g.gameId,rowRevision:g.rowRevision,label:(g.name??'尚無公開名稱')+'／'+g.gameId+(g.management?'／創作者 '+g.creatorAccount:'')+'／管理修訂 '+g.rowRevision,disabled:disabled||Boolean(pending)||bad||g.listed,title:g.listed?'須先由管理員下架':'不可逆柵欄不是物理清理完成'})):[];
  let text='沒有追蹤中的刪除請求。';if(bad)text='本機刪除關聯損壞或無法讀取；不發送新請求。';else if(pending)text='原鍵 '+pending.key+'／目標 '+pending.gameId+'／原修訂 '+pending.expectedRowRevision+(pending.version!==identity?.version?'；帳戶版本已變，不重送。':'；僅明確查詢，不自動重送。');const operation=visible?text:'';
  let result='';if(last){result='本地已建立不可逆刪除柵欄，非物理清理完成。';if(distribution?.state==='confirmed')result+=' 此柵欄分發曾於 '+distribution.visibleAt+' 確認；不證明歷史內容、專屬資源或備份已清除。';else if(distribution)result+=' 柵欄分發待確認。';else result+=' 分發狀態尚未確認。';if(distribution?.recoveryRequired)result+=' 分發處於恢復鎖，本頁不能解除。';}publish({games:rows,operation,result:visible?result:'',message:visible?messageText:'',admin:visible&&identity.role==='admin',locked:disabled,queryDisabled:disabled||!pending||bad,retryDisabled:disabled||!pending||bad||!checked||Boolean(last)||pending.version!==identity?.version,forgetDisabled:disabled||(!pending&&!bad)});
 }
 async function refresh(){const ctx=capture(),value=await api('/api/games');if(!current(ctx))return;if(!value||!Array.isArray(value.games)||value.games.length>10000)throw new Error('遊戲清單無效。');const seen=new Set();games=value.games.map(g=>{if(!g||!uuid(g.gameId)||seen.has(g.gameId)||g.ownerId!==ctx.id||!revision(g.rowRevision)||typeof g.listed!=='boolean'||!nameOK(g.metadata?.name))throw new Error('刪除目標無效。');seen.add(g.gameId);return Object.freeze({gameId:g.gameId,rowRevision:g.rowRevision,listed:g.listed,name:g.metadata.name});});render();}
 async function refreshManaged(){const ctx=capture();if(ctx.role!=='admin')throw new Error('需要管理員權限。');const value=await api('/api/admin/games');if(!current(ctx))return;if(!value||!Array.isArray(value.games)||value.games.length>10000)throw new Error('管理清單無效。');const seen=new Set();games=value.games.map(g=>{if(!g||!uuid(g.gameId)||seen.has(g.gameId)||!revision(g.rowRevision)||typeof g.listed!=='boolean'||typeof g.creatorAccount!=='string'||(g.name!==null&&!nameOK(g.name)))throw new Error('管理刪除目標無效。');seen.add(g.gameId);return Object.freeze({gameId:g.gameId,rowRevision:g.rowRevision,listed:g.listed,name:g.name,creatorAccount:g.creatorAccount,management:true});});render();}
 async function query(){const ctx=capture(),t=pending;if(!t||bad)throw new Error('沒有可查詢的原刪除請求。');const mark=++serial;checked=false;render();const live=()=>current(ctx)&&pending===t&&serial===mark;const path='/api/games/'+t.gameId+'/deletion-operations/'+encodeURIComponent(t.key);try{const value=await api(path);if(!live())return;last=receipt(value,t);render();const value2=await api(path+'/distribution');if(!live())return;const found=observation(value2,t);last=found.result;distribution=found.distribution;message('已核對原刪除柵欄；物理清理仍未獲完成證明。');render();}catch(error){if(live()){if(!last&&error.status===404&&error.code==='DELETE_OPERATION_NOT_FOUND')checked=true;message(error.message+' 保留原鍵，不推定取消或清理完成。');render();}throw error;}}
 async function send(t,ctx){if(sending||!current(ctx)||pending!==t||bad||t.version!==ctx.version)throw new Error('原刪除請求或帳戶已變。');sending=true;checked=false;serial++;render();try{const result=await api('/api/games/'+t.gameId+'/deletion',{gameId:t.gameId,expectedRowRevision:t.expectedRowRevision,confirmationName:t.confirmationName},{key:t.key});if(!current(ctx)||pending!==t)return;last=receipt(result,t);distribution=null;render();try{await query();}catch{/* Local fence remains committed even when observation fails. */}}catch(error){if(current(ctx)&&pending===t){message(error.message+' 原刪除請求保留；先查詢，不另起鍵。');render();}throw error;}finally{sending=false;render();}}
 async function start(g){const ctx=capture();if(pending||bad||g.listed)throw new Error('須先核對原請求，已上架遊戲不能刪除。');let confirmationName=g.name;if(g.management){if(ctx.role!=='admin')throw new Error('需要管理員權限。');const preview=await api('/api/admin/games/'+g.gameId+'/deletion-preview');if(!current(ctx))return;if(!games.includes(g)||pending||bad||!exact(preview,'confirmationName,gameId,rowRevision')||preview.gameId!==g.gameId||preview.rowRevision!==g.rowRevision||!nameOK(preview.confirmationName))throw new Error('刪除確認資料已變或無效，請重新讀取清單。');confirmationName=preview.confirmationName;}const name=prompt('目標 '+g.gameId+'；請完整輸入「'+confirmationName+'」確認不可逆刪除請求。此步僅建立柵欄，不能視為物理清理完成；不刪玩家本地存檔。');if(name===null)return;if(!current(ctx)||!games.includes(g)||pending||bad||g.listed)throw new Error('刪除確認期間帳戶、清單或原操作已變；請重新核對。');if(name!==confirmationName||!nameOK(name))throw new Error('名稱不符，不發送請求。');const t=Object.freeze({actorId:ctx.id,version:ctx.version,gameId:g.gameId,key:crypto.randomUUID(),expectedRowRevision:g.rowRevision,confirmationName:name});store(t);last=null;distribution=null;render();await send(t,ctx);}
 async function retry(){const ctx=capture(),t=pending,mark=serial;if(!t||bad||!checked||last||t.version!==ctx.version)throw new Error('須先確認未知原請求，不能重送已知已提交操作。');if(!confirm('以原鍵、原確認名稱及原修訂重試不可逆刪除請求？404不表示原請求已取消。'))return;if(!current(ctx)||pending!==t||serial!==mark||bad||!checked||last)throw new Error('刪除重試確認已過期，請重新查詢原操作。');await send(t,ctx);}
 function forget(){const ctx=capture(),t=pending,mark=serial;if(!confirm('僅停止本機追蹤？不取消刪除、不撤銷柵欄、不清理服務回執。'))return;if(!current(ctx)||pending!==t||serial!==mark)throw new Error('刪除追蹤確認已過期，原關聯保留。');store(null);last=null;distribution=null;message('已停止本機追蹤，不代表刪除或清理完成。');render();}
 const action=fn=>(...args)=>work(async()=>{const ctx=capture();try{return await fn(...args);}catch(error){if(current(ctx))message(error.message+' 原關聯不自動重送或取消。');throw error;}finally{render();}});
 const actions=Object.freeze({refresh:action(refresh),managed:action(refreshManaged),query:action(query),retry:action(retry),forget:action(forget),start:action((id,rowRevision)=>{const g=games.find(g=>g.gameId===id&&g.rowRevision===rowRevision);if(!g)throw new Error('目標清單已變，請明確重讀。');return start(g);})});
 return Object.freeze({actions,render,setIdentity(value){const u=value?.user,next=u?{id:u.id,version:u.version,role:u.role,mustChangePassword:u.mustChangePassword}:null;if(JSON.stringify(identity)!==JSON.stringify(next)){identity=next;generation++;games=[];message('');restore();}render();},canLeave(forUnload=false){return (!pending&&!bad)||(!forUnload&&confirm('刪除原鍵仍保留於本分頁；離開不取消請求。'));}});
}

export function installGameManagement({api,authority,work,isBusy,publish=()=>{},storage=globalThis.sessionStorage,confirm=text=>globalThis.confirm(text)}) {
 const sessionStorage=storage;
 let messageText='';
 const uuid=v=>typeof v==='string'&&/^[a-f0-9]{8}-[a-f0-9]{4}-4[a-f0-9]{3}-[89ab][a-f0-9]{3}-[a-f0-9]{12}$/.test(v);
 const revision=v=>typeof v==='string'&&v.length<=8192&&/^[1-9][0-9]*$/.test(v);
 const fields=(v,keys)=>v&&typeof v==='object'&&!Array.isArray(v)&&Object.keys(v).sort().join(',')===keys;
 const iso=v=>typeof v==='string'&&/^\d{4}-\d\d-\d\dT\d\d:\d\d:\d\d\.\d{3}Z$/.test(v)&&Number.isFinite(Date.parse(v))&&new Date(v).toISOString()===v;
 const actions=['list','unlist','clear-restriction'];
 let identity=null,generation=0,games=[],pending=null,checked=false,badJournal=false,last=null,distribution,querySerial=0,sending=false;
 const allowed=()=>identity?.user.role==='admin'&&!identity.user.mustChangePassword;
 const message=text=>{messageText=text;};
 const journalKey=id=>'editor-management-operation:'+id;
 function tuple(v){return fields(v,'action,actorId,expected,gameId,key,version')&&uuid(v.actorId)&&uuid(v.gameId)&&Number.isSafeInteger(v.version)&&v.version>0&&typeof v.key==='string'&&/^[A-Za-z0-9_-]{16,128}$/.test(v.key)&&actions.includes(v.action)&&revision(v.expected);}
 function current(ctx){const live=authority();return allowed()&&live?.user.id===ctx.id&&live.user.version===ctx.version&&live.user.role==='admin'&&!live.user.mustChangePassword&&generation===ctx.generation;}
 function capture(){const ctx={id:identity?.user.id,version:identity?.user.version,generation};if(!current(ctx))throw new Error('請先核對目前管理員帳戶。');return ctx;}
 function store(value){
  const key=journalKey(identity.user.id),text=value?JSON.stringify(value):null;
  try{if(value)sessionStorage.setItem(key,text);else sessionStorage.removeItem(key);if(sessionStorage.getItem(key)!==text)throw new Error('journal readback');}
  catch{badJournal=true;render();throw new Error('無法保留管理操作關聯；不發送新請求。');}
  pending=value;checked=false;badJournal=false;querySerial++;
 }
 function restore(){pending=null;checked=false;badJournal=false;last=null;distribution=undefined;querySerial++;if(!allowed())return;try{const raw=sessionStorage.getItem(journalKey(identity.user.id));if(raw!==null){if(raw.length>20000)throw new Error('journal budget');let v;try{v=JSON.parse(raw);}catch{throw new Error('journal JSON');}if(!tuple(v)||v.actorId!==identity.user.id)throw new Error('journal tuple');pending=Object.freeze(v);}}catch{badJournal=true;}}
 function receipt(value,t){
  if(t.action==='list'){
   if(!fields(value,'createdAt,currentReleaseId,draftRevision,gameId,listed,metadata,modifiedAt,nextOrdinal,ownerId,restricted,rowRevision')||value.gameId!==t.gameId||!uuid(value.ownerId)||!uuid(value.currentReleaseId)||!revision(value.draftRevision)||typeof value.nextOrdinal!=='string'||!/^1\.(0|[1-9][0-9]{0,30})$/.test(value.nextOrdinal)||value.listed!==true||value.restricted!==false||!fields(value.metadata,'introduction,name')||typeof value.metadata.name!=='string'||typeof value.metadata.introduction!=='string'||!iso(value.createdAt)||!iso(value.modifiedAt))throw new Error('上架回執格式無效；保留原鍵。');
  }else if(!fields(value,'action,eventAt,gameId,listed,restricted,rowRevision')||value.gameId!==t.gameId||value.action!==t.action||value.listed!==false||value.restricted!==(t.action==='unlist')||!iso(value.eventAt))throw new Error('管理回執格式無效；保留原鍵。');
  if(!revision(value.rowRevision))throw new Error('管理回執修訂無效。');
  const n=BigInt(value.rowRevision),old=BigInt(t.expected);if(n!==old&&n!==old+1n)throw new Error('管理回執修訂不符；保留原關聯。');
  if(last&&Object.keys(value).some(k=>k==='metadata'?['name','introduction'].some(p=>value.metadata[p]!==last.metadata?.[p]):value[k]!==last[k]))throw new Error('原管理回執不一致。');
  return Object.freeze({...value,...(t.action==='list'?{metadata:Object.freeze({...value.metadata})}:{})});
 }
 function observation(value,t){
  if(!fields(value,'distribution,gameId,result,version')||value.version!==1||value.gameId!==t.gameId)throw new Error('管理分發回執無效。');
  const result=receipt(value.result,t),d=value.distribution;
  if(d!==null&&(!fields(d,'gameId,generation,kind,operationId,recoveryRequired,state,visibleAt')||d.gameId!==t.gameId||!uuid(d.operationId)||!Number.isSafeInteger(d.generation)||d.generation<1||d.kind!==(t.action==='list'?'listing':'restriction')||typeof d.recoveryRequired!=='boolean'||!['pending','confirmed'].includes(d.state)||(d.state==='pending'?d.visibleAt!==null:!iso(d.visibleAt))))throw new Error('管理分發狀態無效；不推定成功。');
  if(distribution!==undefined&&(distribution===null?d!==null:d===null||['operationId','generation','kind'].some(k=>distribution[k]!==d[k])))throw new Error('原管理分發關聯不一致。');
  if(d===null&&result.rowRevision!==t.expected)throw new Error('變更操作不能沒有分發關聯。');
  if(d!==null&&result.rowRevision===t.expected)throw new Error('無變更操作不能新增分發。');
  return {result,distribution:d===null?null:Object.freeze({...d})};
 }
 function list(value){if(!value||!Array.isArray(value.games)||value.games.length>10000)throw new Error('管理清單格式無效。');const seen=new Set();return value.games.map(g=>{if(!g||!uuid(g.gameId)||seen.has(g.gameId)||typeof g.creatorAccount!=='string'||(g.name!==null&&typeof g.name!=='string')||(g.introduction!==null&&typeof g.introduction!=='string')||(g.currentReleaseId!==null&&!uuid(g.currentReleaseId))||typeof g.listed!=='boolean'||typeof g.restricted!=='boolean'||!revision(g.rowRevision)||typeof g.createdAt!=='string'||typeof g.modifiedAt!=='string')throw new Error('管理清單欄位無效。');seen.add(g.gameId);return Object.freeze({...g});});}
 function render(){
  const visible=current({id:identity?.user.id,version:identity?.user.version,generation}),disabled=!visible||isBusy()||sending;
  const rows=visible?games.map(g=>({gameId:g.gameId,rowRevision:g.rowRevision,label:(g.name??(g.currentReleaseId?'已有正式版本（未有公開名稱）':'尚無正式版本'))+' — '+(g.introduction??'無簡介')+'／'+g.gameId+'／創作者 '+g.creatorAccount+'／創建 '+g.createdAt+'／修改 '+g.modifiedAt+'／本地'+(g.listed?'上架標記':'下架標記')+'／'+(g.restricted?'管理限制':'無管理限制')+'／管理修訂 '+g.rowRevision,actions:actions.map(action=>({action,label:action==='list'?'上架正式版本':action==='unlist'?(g.listed?'管理下架並限制':'設管理限制（保持下架）'):'解除管理限制（不自動上架）',disabled:disabled||Boolean(pending)||badJournal||(action==='clear-restriction'&&!g.restricted)||(action==='list'&&(!g.currentReleaseId||g.restricted||g.listed))}))})):[];
  let text='沒有追蹤中的管理操作。';if(badJournal)text='本機管理關聯無法核對；不發送新操作。';else if(pending)text='原操作 '+pending.key+'／目標 '+pending.gameId+'／'+pending.action+'／原管理修訂 '+pending.expected+(pending.version!==identity?.user.version?'；帳戶版本已變，不重送舊請求。':'')+(checked?'；GET尚未找到回執，不表示已取消。':'；可只讀查詢，不自動重送。');const operation=visible?text:'';
  let result='';if(last){result='本地已提交原操作（歷史事實，不是目前狀態）：'+pending.action+'／目標 '+last.gameId+'／回執管理修訂 '+last.rowRevision+'。';
   if(distribution===null)result+=' 此次未變更，沒有新增分發；不代表目前遠端狀態。';
   else if(distribution?.state==='confirmed')result+=' 此次分發曾於 '+distribution.visibleAt+' 確認；不是目前上架或最新遠端狀態。';
   else result+=' 分發待確認／狀態不可用；不推定玩家已可見。';
   if(distribution?.recoveryRequired)result+=' 分發處於恢復鎖，本頁不能解除。';
  }publish({games:rows,operation,result:visible?result:'',message:visible?messageText:'',locked:disabled,queryDisabled:disabled||!pending||badJournal,retryDisabled:disabled||!pending||badJournal||!checked||Boolean(last)||pending.version!==identity?.user.version,forgetDisabled:disabled||(!pending&&!badJournal)});
 }
 async function refresh(){const ctx=capture();const result=await api('/api/admin/games');if(!current(ctx))return;games=list(result);render();}
 async function query(){
  const ctx=capture(),t=pending;if(!t||badJournal)throw new Error('沒有可核對的原操作關聯。');const serial=++querySerial;checked=false;render();
  const live=()=>current(ctx)&&pending===t&&serial===querySerial;
  const path='/api/admin/games/'+t.gameId+'/management-operations/'+encodeURIComponent(t.key);
  try{
   const value=await api(path);if(!live())return;last=receipt(value,t);render();
   // Only the permanent receipt endpoint's explicit absence can permit retry.
   const value2=await api(path+'/distribution');if(!live())return;const found=observation(value2,t);last=found.result;distribution=found.distribution;message('已核對原操作；本地提交、歷史分發確認和目前狀態不同。');render();
  }catch(error){if(live()){if(!last&&error.status===404&&error.code==='MANAGEMENT_OPERATION_NOT_FOUND')checked=true;message(error.message+' 原關聯保留；不推定成功、失敗或取消。');render();}throw error;}
 }
 async function send(t,ctx){
  if(sending||!current(ctx)||pending!==t||badJournal||t.version!==ctx.version)throw new Error('原管理關聯或帳戶已變。');sending=true;checked=false;querySerial++;render();
  const path='/api/games/'+t.gameId+(t.action==='clear-restriction'?'/restriction/clear':'/listing');
  const body=t.action==='list'?{action:'list',expectedRowRevision:t.expected}:t.action==='unlist'?{action:'unlist'}:{};
  try{const value=await api(path,body,{key:t.key,revision:t.expected});if(!current(ctx)||pending!==t)return;last=receipt(value,t);distribution=undefined;message('本地已提交，正在只讀核對分發。');render();try{await query();}catch{/* A failed observation does not undo a local commit. */}}
  catch(error){if(current(ctx)&&pending===t){message(error.message+' 原關聯保留；先查詢，不自動重試或另起鍵。');render();}throw error;}
  finally{sending=false;render();}
 }
