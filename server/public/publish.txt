// 上架面板控制器（copy 单槽 API）：行 = 游戏 × submission 状态机。
// 检验走 POST /copy/validate（落库），上架走 POST /copy/submission（ticket 一次性 + 说明必填），
// 提交后只轮询 GET submission 的 slot（A 路线 190s 窗口）。不设 revision/operation-key/长连接。
export function installPublish({ api, authority, work, isBusy, publish: emit = () => {}, gameContext, storage = globalThis.sessionStorage, confirm = text => globalThis.confirm(text) }) {
 const uuid = v => typeof v === 'string' && /^[a-f0-9]{8}-[a-f0-9]{4}-4[a-f0-9]{3}-[89ab][a-f0-9]{3}-[a-f0-9]{12}$/.test(v);
 const ordinal = v => typeof v === 'string' && /^1\.(0|[1-9][0-9]{0,30})$/.test(v);
 const digest = v => typeof v === 'string' && /^[a-f0-9]{64}$/.test(v);
 const iso = v => typeof v === 'string' && /^\d{4}-\d\d-\d\dT\d\d:\d\d:\d\d\.\d{3}Z$/.test(v) && Number.isFinite(Date.parse(v)) && new Date(v).toISOString() === v;
 const errors = { COPY_NOT_FOUND: '副本不存在，请先新建游戏。', VALIDATION_STALE: '检验票已失效（副本已变更或票已使用），请重新检验。', VALIDATION_TICKET: '检验票无效，请重新检验。', PUBLISH_TOO_SOON: '提交太频繁，请稍后再试。', PUBLISH_DESCRIPTION: '发布说明须为 1..2000 字。', GAME_NOT_FOUND: '游戏不存在。', PUBLISH_CHAPTER_REFERENCES: '整游戏校验未通过：有章节引用不到已删除的基础资料（据点或武将库）；下方列出受影响的章。' };
 const SUBMIT_POLL_MS = 5000, SUBMIT_POLL_BUDGET_MS = 190000;
 let identity = null, generation = 0;
 let rows = [], listed = false, slots = new Map(), tickets = new Map();
 let description = '', messageText = '', resultText = '', lastCheck = null;
 let slotWatch = null;
 let sending = false;
 const allowed = () => Boolean(identity?.user) && !identity.user.mustChangePassword && ['author', 'admin'].includes(identity.user.role);
 const stateLabels = { draft: '草稿', pending: '待审核', listed: '已上架' };
 function ticket() { const actor = identity, serial = ++generation; return () => allowed() && identity === actor && generation === serial; }
 function current(ctx) { const live = authority(); return allowed() && live?.user.id === ctx.id && live.user.version === ctx.version && !live.user.mustChangePassword && generation === ctx.generation; }
 function capture() { const ctx = { id: identity?.user.id, version: identity?.user.version, generation }; if (!current(ctx)) throw new Error('请重新核对登入的帐户。'); return ctx; }
 const message = text => { messageText = text; };
 function checkSubmission(value, gameId) {
  if (!value || typeof value !== 'object' || Array.isArray(value) || value.gameId !== gameId || !['draft', 'pending', 'listed'].includes(value.state)) throw new Error('上架状态回应无效。');
  if (value.copyDigest !== null && !digest(value.copyDigest)) throw new Error('上架状态回应无效。');
  if (value.validation !== null && (!value.validation || typeof value.validation !== 'object' || !['passed', 'failed'].includes(value.validation.state) || !iso(value.validation.checkedAt) || !Array.isArray(value.validation.problems))) throw new Error('上架状态回应无效。');
  if (value.slot !== null && (!value.slot || typeof value.slot !== 'object' || !['empty', 'ready', 'submitted', 'approved'].includes(value.slot.state))) throw new Error('上架状态回应无效。');
  if ((value.version !== null && !ordinal(value.version)) || !ordinal(value.nextVersion) || typeof value.publishEnabled !== 'boolean' || (value.blockedReason !== null && typeof value.blockedReason !== 'string')) throw new Error('上架状态回应无效。');
  return Object.freeze(value);
 }
 function validationLabel(sub) {
  const validation = sub?.validation ?? null;
  if (!validation) return '未检验';
  if (validation.state === 'failed') return '检验失败';
  if (sub.blockedReason === '副本已变更，请重新检验。') return '检验已过期';
  return '检验成功';
 }
 function validationHint(sub) {
  const validation = sub?.validation ?? null;
  if (!validation) return '尚未检验：先按「校验」跑全量检验，通过后才能送出上架申请';
  if (validation.state === 'failed') return '上次检验未通过；依红字修正后重新检验';
  return '';
 }
 // 检验清单（常驻两项，code 与服务端 validateWhole 的问题码一一对应；未知码追加在后）。
 // 未检验时 done=false；新一轮开始即清掉上轮结论，面板据 checking 显示检查中。
 const validationChecklist = Object.freeze([
  { key: 'CHAPTER_REFERENCE', code: 'CHAPTER_REFERENCE', title: '章节引用完整性' },
  { key: 'VALIDATION_FAILED', code: 'VALIDATION_FAILED', title: '游戏源与地图编译' },
 ]);
 function layerRows() {
  const groups = new Map();
  for (const problem of lastCheck?.problems ?? []) {
   const code = typeof problem?.code === 'string' && problem.code ? problem.code : 'UNKNOWN';
   const bucket = groups.get(code) ?? [];
   bucket.push({ chapterId: typeof problem?.chapterId === 'string' ? problem.chapterId : '?', name: typeof problem?.name === 'string' && problem.name ? problem.name : (typeof problem?.chapterId === 'string' ? problem.chapterId : '?'), detail: String(problem?.detail ?? '') });
   groups.set(code, bucket);
  }
  const rows = validationChecklist.map(known => {
   const failures = groups.get(known.code) ?? [];
   groups.delete(known.code);
   if (!lastCheck) return { ...known, done: false, ok: true, failures: [] };
   return { ...known, done: true, ok: failures.length === 0, failures };
  });
  for (const [code, failures] of groups) rows.push({ key: code, code, title: code, done: true, ok: failures.length === 0, failures });
  return rows;
 }
 function render() {
  const ctx = identity ? { id: identity.user.id, version: identity.user.version, generation } : null;
  const visible = Boolean(ctx && current(ctx)), disabled = !visible || isBusy();
  const list = visible ? rows.map(row => {
   const sub = slots.get(row.gameId) ?? null;
   return { gameId: row.gameId, name: row.name, introduction: row.introduction || '（无说明）', version: sub?.version ?? '—', nextVersion: sub?.nextVersion ?? '—',
    state: sub ? stateLabels[sub.state] : '未知', stateKey: sub?.state ?? 'unknown', rejected: false, createdAt: row.createdAt, stateTime: sub?.slot?.submittedAt ?? sub?.validation?.checkedAt ?? sub?.copyUpdatedAt ?? '—',
    validation: sub ? validationLabel(sub) : '未读取', validationTime: sub?.validation ? new Date(sub.validation.checkedAt).toLocaleString() : '—',
    validationKey: sub?.validation ? sub.validation.state : 'none', validationState: sub?.validation ? sub.validation.state : 'none',
    publishDisabled: disabled || !sub || !sub.publishEnabled || ![...description.trim()].length,
    publishHint: !sub ? '正在读取上架状态…' : !sub.publishEnabled ? (sub.blockedReason ?? '暂不可提交') : ![...description.trim()].length ? '请先填写发布说明（1..2000 字）' : '校验整个游戏；通过后版本号自动为 ' + sub.nextVersion + ' 并提交管理员审核',
    publishTitle: sub ? validationHint(sub) || sub.blockedReason || '' : '' };
  }) : [];
  emit({ rows: list, description, descriptionCount: [...description].length, message: visible ? messageText : '', result: visible ? resultText : '',
   validationLayers: visible ? layerRows() : [], validationChecked: visible && Boolean(lastCheck), validationPassed: visible && Boolean(lastCheck) && !lastCheck.problems.length,
   steps: [], stepsFailed: false, locked: disabled, refreshDisabled: !visible || isBusy(), checkDisabled: !visible || isBusy() || sending, phase: '' });
 }
 async function readSlot(ctx, gameId) {
  const value = checkSubmission(await api('/api/games/' + gameId + '/copy/submission'), gameId);
  if (!current(ctx)) return null;
  slots.set(gameId, value);
  return value;
 }
 async function load(ctx) {
  const value = await api('/api/games');
  if (!current(ctx)) return false;
  if (!Array.isArray(value?.games)) throw new Error('游戏清单回应无效。');
  const ids = new Set();
  rows = value.games.map(game => {
   if (typeof game?.gameId !== 'string' || !uuid(game.gameId) || ids.has(game.gameId) || typeof game?.metadata?.name !== 'string') throw new Error('游戏清单回应无效。');
   ids.add(game.gameId);
   return { gameId: game.gameId, name: game.metadata.name, introduction: game.metadata.introduction ?? '', createdAt: game.createdAt ?? '—' };
  });
  listed = true;
  for (const row of rows) {
   try { await readSlot(ctx, row.gameId); } catch (error) { if (!current(ctx)) return false; if (error?.status === 404 || error?.code === 'COPY_NOT_FOUND') slots.delete(row.gameId); else throw error; }
   if (!current(ctx)) return false;
  }
  return true;
 }
 function stopSlotWatch() { if (slotWatch) { clearInterval(slotWatch.timer); slotWatch = null; } }
 function startSlotWatch(ctx, gameId) {
  stopSlotWatch();
  const started = Date.now();
  const watch = { gameId, timer: setInterval(async () => {
   if (!current(ctx) || slotWatch !== watch) { stopSlotWatch(); return; }
   try { await readSlot(ctx, gameId); } catch { stopSlotWatch(); render(); return; }
   if (!current(ctx) || slotWatch !== watch) { stopSlotWatch(); return; }
   const sub = slots.get(gameId);
   render();
   if (sub?.slot?.state !== 'submitted' || Date.now() - started > SUBMIT_POLL_BUDGET_MS) {
    stopSlotWatch();
    if (sub?.slot?.state !== 'submitted') message('上架申请状态已更新：' + (sub.slot.state === 'approved' ? '管理员已同意。' : '槽状态 ' + sub.slot.state + '。'));
    else message('提交已超过 190 秒仍待审核；以上架页重读的状态为准，不自动重送。');
    render();
   }
  }, SUBMIT_POLL_MS) };
  slotWatch = watch;
 }
 async function refresh() { const ctx = capture(); stopSlotWatch(); lastCheck = null; message(''); resultText = ''; await load(ctx); render(); return '上架清单已更新。'; }
 // 唯读检验：POST 落库，票据随通过回应下发；面板遮罩按 code 分组逐项列出红字。
 async function check(gameId) {
  const ctx = capture();
  const row = rows.find(entry => entry.gameId === gameId);
  if (!row) throw new Error('上架清单已变，请重新整理。');
  lastCheck = null; render();
  // B8 逐章清单（只读，走既有 draft 路由）：有草稿链的游戏先列逐章引用缺口，
  // 先于 copy 全量检验；纯 copy 游戏没有草稿链 blob（BLOB_NOT_FOUND）或草稿子系统
  // 未配置（DRAFT_NOT_CONFIGURED）时放行，以 copy/validate 全量检验为准
  // （它同样跑 chapterReferenceReport）。其它错误一律抛出，不吞。
  const flattenChapters = chapters => {
   const list = [];
   for (const row of chapters ?? []) for (const problem of row?.problems ?? []) list.push({ code: 'CHAPTER_REFERENCE', chapterId: row.chapterId, name: row.chapterName ?? row.chapterId, detail: `[${row.chapterName ?? row.chapterId}] ${problem.code}: ${problem.detail}` });
   return list;
  };
  const readChapterReferences = async () => {
   try { return flattenChapters((await api('/api/games/' + gameId + '/draft/chapter-references', {}))?.chapters); }
   catch (error) {
    if (error?.code === 'PUBLISH_CHAPTER_REFERENCES') return flattenChapters(error.detail?.chapters);
    if (error?.code === 'DRAFT_NOT_CONFIGURED' || error?.code === 'BLOB_NOT_FOUND') return [];
    throw error;
   }
  };
  const chapterGaps = await readChapterReferences();
  if (!current(ctx)) return;
  if (chapterGaps.length) {
   lastCheck = { gameId, problems: chapterGaps, checkedAt: new Date().toISOString() };
   tickets.delete(gameId);
   render();
   const blocked = new Error(errors.PUBLISH_CHAPTER_REFERENCES); blocked.code = 'PUBLISH_CHAPTER_REFERENCES'; throw blocked;
  }
  // POST 必须带空对象体：api() 无 body 即发 GET，而检验只有 POST 路由（漏体会 404 NOT_FOUND）。
  // 无副本行时先 ensure（幂等，只种一次）再验一次；新游戏首验一键走通。
  const validateOrEnsure = async gameId => {
    try { return await api('/api/games/' + gameId + '/copy/validate', {}); }
    catch (error) {
      if (error?.code !== 'COPY_NOT_FOUND') throw error;
      await api('/api/games/' + gameId + '/copy/ensure', {});
      return await api('/api/games/' + gameId + '/copy/validate', {});
    }
  };
  const value = await validateOrEnsure(gameId);
  if (!current(ctx)) return;
  if (!value || value.gameId !== gameId || !['passed', 'failed'].includes(value.state) || !digest(value.copyDigest) || !Array.isArray(value.problems)) throw new Error('检验回应无效。');
  lastCheck = { gameId, problems: value.problems, checkedAt: value.checkedAt ?? null };
  if (value.state === 'passed' && typeof value.ticket === 'string') tickets.set(gameId, { ticket: value.ticket, digest: value.copyDigest });
  else tickets.delete(gameId);
  try { await readSlot(ctx, gameId); } catch { /* 清单留待手动刷新；本次检验结果仍在下方 */ }
  if (!current(ctx)) return;
  message(value.problems.length ? '检验失败！' : '检验成功！' + (slots.get(gameId)?.publishEnabled ? '可以送出上架申请（仍须管理员同意）。' : '暂不可提交：' + (slots.get(gameId)?.blockedReason ?? '')));
  render();
  return value.problems.length ? '检验未通过，请看下方清单。' : '检验完成。';
 }
 function setDescription(text) {
  if (typeof text !== 'string') return;
  description = [...text].length > 2000 ? [...text].slice(0, 2000).join('') : text;
  render();
 }
 // 一笔上架只送一次 POST：票据一次性，说明必填，节流由服务端判定（429 照原文抛出）。
 async function submit(gameId) {
  const ctx = capture();
  const text = description.trim().normalize('NFC');
  if ([...text].length < 1 || [...text].length > 2000) throw new Error('发布说明须为 1..2000 字。');
  let row = rows.find(entry => entry.gameId === gameId);
  if (!row) throw new Error('上架清单已变，请重新整理。');
  if (sending) throw new Error('正在提交，请稍候。');
  sending = true; render();
  try {
  let sub = slots.get(gameId) ?? null, tk = tickets.get(gameId) ?? null;
  if (!tk || !sub || tk.digest !== sub.copyDigest) {
   await check(gameId);
   if (!current(ctx)) return;
   sub = slots.get(gameId) ?? null; tk = tickets.get(gameId) ?? null;
   row = rows.find(entry => entry.gameId === gameId) ?? row;
  }
  if (!tk || !sub || tk.digest !== sub.copyDigest) throw new Error('检验未通过，无法送出上架申请。');
  const value = await api('/api/games/' + gameId + '/copy/submission', { ticket: tk.ticket, description: text });
  if (!current(ctx)) return;
  if (!value || value.gameId !== gameId || value.state !== 'submitted') throw new Error('上架申请回应无效。');
  tickets.delete(gameId); description = ''; resultText = '已提交上架申请：版本待管理员同意后确定。';
  try { await readSlot(ctx, gameId); } catch { /* 轮询会重读；本次提交已成立 */ }
  if (!current(ctx)) return;
  startSlotWatch(ctx, gameId);
  render();
  return '已提交上架申请。';
  } finally { sending = false; render(); }
 }
 const action = fn => (...args) => work(async () => { const ctx = capture(); try { return await fn(...args); } catch (error) { if (current(ctx)) message((errors[error.code] ?? error.message) + ' 本次未提交；修正后可重新送出上架申请。'); throw error; } finally { render(); } });
 const actions = Object.freeze({ refresh: action(refresh), check: action(check), publish: action(submit), setDescription });
 return Object.freeze({ actions, render,
  setIdentity(value) { const next = value?.user ? { user: { id: value.user.id, version: value.user.version, role: value.user.role, mustChangePassword: value.user.mustChangePassword } } : null; const changed = JSON.stringify(identity) !== JSON.stringify(next); identity = next; if (changed) { generation++; rows = []; listed = false; slots = new Map(); tickets = new Map(); description = ''; message(''); resultText = ''; lastCheck = null; stopSlotWatch(); } render(); },
  selectedGame: () => null,
  canAdoptGame(id) { return allowed() && uuid(id); },
  adoptGame() { render(); },
  canLeave() { return true; } });
}
