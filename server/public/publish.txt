// B2 上架管理 controller（Web 產品決定 2026-11）：一遊戲一行；「上架」＝整遊戲校驗
// （既有 publish 管線：validateGameSource＋compile）通過後版本號自動 +1 並提交管理員
// 審核；「刪除」＝不可逆刪除柵欄（不是物理清理，也不刪玩家本地存檔）。
// 三個顯示態由服務端既有資料推導（草稿／待上架／已上架），本檔只做展示與操作追蹤。
// Browser journals never grant service authority.
export function installPublish({ api, authority, work, isBusy, publish: emit = () => {}, gameContext, storage = globalThis.sessionStorage, confirm = text => globalThis.confirm(text) }) {
 const uuid = v => typeof v === 'string' && /^[a-f0-9]{8}-[a-f0-9]{4}-4[a-f0-9]{3}-[89ab][a-f0-9]{3}-[a-f0-9]{12}$/.test(v);
 const revision = v => typeof v === 'string' && /^[1-9][0-9]{0,30}$/.test(v);
 const ordinal = v => typeof v === 'string' && /^1\.(0|[1-9][0-9]{0,30})$/.test(v);
 const digest = v => typeof v === 'string' && /^[a-f0-9]{64}$/.test(v);
 const iso = v => typeof v === 'string' && /^\d{4}-\d\d-\d\dT\d\d:\d\d:\d\d\.\d{3}Z$/.test(v) && Number.isFinite(Date.parse(v)) && new Date(v).toISOString() === v;
 const fields = (v, keys) => v && typeof v === 'object' && !Array.isArray(v) && Object.keys(v).sort().join(',') === [...keys].sort().join(',');
 // A 路線第一版：上架回執可選擇性多帶一個分發描述子，欄位集只放行這一個額外鍵。
 const fieldsOptional = (v, keys, optional) => v && typeof v === 'object' && !Array.isArray(v) && Object.keys(v).sort().join(',') === (optional !== undefined && Object.hasOwn(v, optional) ? [...keys, optional] : [...keys]).sort().join(',');
 const stateLabels = { draft: '草稿', pending: '待上架', listed: '已上架' };
 // 「版本號 +1」只做顯示推導；實際序號由服務端 release 帳本配發。
 const nextVersion = v => v === null ? '1.0' : v.slice(0, v.lastIndexOf('.')) + '.' + String(BigInt(v.slice(v.lastIndexOf('.') + 1)) + 1n);
 const errors = { DRAFT_TOPOLOGY_BROKEN: '仍有據點移動意向，不能上架；請先核對並處理原意向。', PLAY_CITY_VIEW_MISSING: '章節引用了固定資源庫未提供的據點圖片；請核對章節資源與已保存草稿，未補齊前不能上架。', PLAY_PORTRAIT_CORRUPT: '武將畫像引用無效或缺少可凍結的依賴；請核對已保存章節的畫像資料。', PUBLISH_STALE: '草稿已變更，請重新整理清單後再上架。', PUBLISH_NAME: '遊戲名稱無效（去除首尾空白及NFC正規化後，須為1至8個Unicode碼點）。', DISTRIBUTION_GAME_PENDING: '此遊戲仍有待確認分發，不能上架下一版。', DISTRIBUTION_NOT_CONFIGURED: '服務尚未配置分發狀態；不推定已公開。', DISTRIBUTION_STATUS_UNAVAILABLE: '原上架缺少可核對的分發關聯；不推定成功或失敗。', GAME_DELETING: '遊戲已進入刪除流程，不能繼續此操作。', DELETE_CONFIRMATION: '刪除確認名稱與目前遊戲名稱不符，請重新整理清單。', PUBLISH_CHAPTER_REFERENCES: '整遊戲校驗未通過：有章節引用不到已刪除的基礎資料（據點或武將庫）；下方列出受影響的章。' };

 // 阻塞遮罩的第二行：每個等待點都先寫清楚現在在做什麼，不要讓人只知道「在忙」。
 let phase = '';
 const allowed = () => Boolean(identity?.user) && !identity.user.mustChangePassword && ['author', 'admin'].includes(identity.user.role);
 const message = text => { messageText = text; };
 const journalKey = id => 'editor-submission-operation:' + id;
 function tuple(v) {
  if (!v || typeof v !== 'object' || Array.isArray(v)) return false;
  if (!uuid(v.actorId) || !uuid(v.gameId) || !Number.isSafeInteger(v.version) || v.version < 1 || typeof v.key !== 'string' || !/^[A-Za-z0-9_-]{16,128}$/.test(v.key)) return false;
  if (v.kind === 'publish') return fields(v, ['actorId', 'gameId', 'key', 'kind', 'name', 'version']) && typeof v.name === 'string' && v.name.isWellFormed() && [...v.name].length >= 1 && [...v.name].length <= 8;
  if (v.kind === 'delete') return fields(v, ['actorId', 'confirmationName', 'expectedRowRevision', 'gameId', 'key', 'kind', 'version']) && typeof v.confirmationName === 'string' && v.confirmationName.isWellFormed() && [...v.confirmationName].length <= 8 && revision(v.expectedRowRevision);
  return false;
 }
 function current(ctx) { const live = authority(); return allowed() && live?.user.id === ctx.id && live.user.version === ctx.version && !live.user.mustChangePassword && generation === ctx.generation; }
 function capture() { const ctx = { id: identity?.user.id, version: identity?.user.version, generation }; if (!current(ctx)) throw new Error('請重新核對登入的帳戶。'); return ctx; }
 function store(value) {
  const key = journalKey(identity.user.id), text = value ? JSON.stringify(value) : null;
  try { if (value) storage.setItem(key, text); else storage.removeItem(key); if (storage.getItem(key) !== text) throw new Error('journal readback'); }
  catch { badJournal = true; render(); throw new Error('無法保留操作關聯；不發送新請求。'); }
  pending = value; checked = false; badJournal = false; querySerial++;
 }
 function restore() {
  pending = null; checked = false; badJournal = false; result = null; distribution = null; querySerial++;
  if (!allowed()) return;
  try { const text = storage.getItem(journalKey(identity.user.id)); if (text !== null) { if (text.length > 4096) throw new Error('journal budget'); let value; try { value = JSON.parse(text); } catch { throw new Error('journal JSON'); } if (!tuple(value) || value.actorId !== identity.user.id) throw new Error('journal tuple'); pending = Object.freeze(value); } }
  catch { badJournal = true; }
 }
 function rowList(value) {
  if (!value || !Array.isArray(value.games) || value.games.length > 10000) throw new Error('上架清單格式無效。');
  const seen = new Set();
  return value.games.map(row => {
   if (!row || !uuid(row.gameId) || seen.has(row.gameId) || typeof row.name !== 'string' || !row.name.isWellFormed() || [...row.name].length < 1 || [...row.name].length > 8 || typeof row.introduction !== 'string' || !Object.hasOwn(stateLabels, row.state) || (row.version !== null && !ordinal(row.version)) || (row.state === 'draft') !== (row.version === null) || (row.stateTime !== null && !iso(row.stateTime)) || !iso(row.createdAt) || typeof row.restricted !== 'boolean' || typeof row.rejected !== 'boolean' || (row.rejected && row.state !== 'pending') || !revision(row.rowRevision)) throw new Error('上架清單欄位無效。');
   seen.add(row.gameId); return Object.freeze({ ...row });
  });
 }
 // Permanent publish receipt: the release pin, not a visibility claim.
 function publishReceipt(value, t) {
  if (!fieldsOptional(value, ['gameId', 'name', 'revision', 'sourceDigest', 'chapters', 'publishedAt', 'ordinal'], 'distribution') || value.gameId !== t.gameId || value.name !== t.name || !ordinal(value.ordinal) || !revision(value.revision) || !digest(value.sourceDigest) || !iso(value.publishedAt) || !Array.isArray(value.chapters) || !value.chapters.length || value.chapters.some(id => typeof id !== 'string' || !id.length) || new Set(value.chapters).size !== value.chapters.length) throw new Error('上架回執格式無效；保留同一個請求。');
  // 逆向寫回每一個欄位：可選的分發描述子不得混進永久上架回執。
  return Object.freeze({ gameId: value.gameId, name: value.name, revision: value.revision, sourceDigest: value.sourceDigest, chapters: Object.freeze([...value.chapters]), publishedAt: value.publishedAt, ordinal: value.ordinal });
 }
 // C 批：物理刪除回執多一個選填的回收結果（在架指標移除＋R2 前綴位元組回收）。
 const reclaimed = value => value === undefined || value !== null && typeof value === 'object' && !Array.isArray(value) &&
  ['completed', 'pending'].includes(value.state) && Number.isSafeInteger(value.objects) && value.objects >= 0 &&
  Object.keys(value).every(key => ['state', 'code', 'objects', 'pointer'].includes(key)) &&
  (value.state === 'completed' ? ['rewritten', 'absent', 'already-absent'].includes(value.pointer) && value.code === undefined : typeof value.code === 'string' && value.code.length >= 1 && value.code.length <= 200 && !/[\x00-\x1f\x7f]/.test(value.code) && value.pointer === undefined);
 const reclaimText = value => !value ? '' : value.state === 'completed'
  ? ' 在架指標與 R2 位元組已回收。'
  : value.code === 'RECLAIMER_UNCONFIGURED' ? ' R2 回收未設定，位元組可能仍留在儲存。'
  : ' R2 回收未完成（' + value.code + '），可稍後重試。';
 function purgeReceipt(value) {
  if ((!fields(value, ['gameId', 'purgedAt', 'reclamation', 'removed', 'state', 'tables', 'version']) && !fields(value, ['gameId', 'purgedAt', 'reclamation', 'removed', 'retirement', 'state', 'tables', 'version'])) || value.version !== 1 || value.state !== 'purged' || !uuid(value.gameId) || !Number.isSafeInteger(value.tables) || typeof value.purgedAt !== 'string' || !value.removed || typeof value.removed !== 'object' || !reclaimed(value.reclamation)) throw new Error('徹底刪除回執無效。');
  if (Number(value.removed.content_games) !== 1) throw new Error('徹底刪除回執無效：遊戲列數不符。');
  return Object.freeze({ ...value });
 }
 function deleteReceipt(value, t) {
  if (!fields(value, ['createdAt', 'gameId', 'rowRevision', 'state']) || value.gameId !== t.gameId || value.state !== 'fenced' || !iso(value.createdAt) || !revision(value.rowRevision) || BigInt(value.rowRevision) !== BigInt(t.expectedRowRevision) + 1n) throw new Error('刪除回執無效；保留同一個請求。');
  return Object.freeze({ ...value });
 }
 // 分發狀態（A 路線第一版）：GET 狀態與同步上架回執共用同一套欄位與語意，
 // 只有 confirmed 才代表這次代次曾被觀測為公開。releaseId 由呼叫端指定。
 function distributionStatus(d, gameId, releaseId) {
  if (!fields(d, ['gameId', 'operationId', 'generation', 'kind', 'state', 'visibleAt', 'releaseDigest', 'recoveryRequired']) || d.gameId !== gameId || d.operationId !== releaseId || d.kind !== 'publish' || !Number.isSafeInteger(d.generation) || d.generation < 1 || (d.releaseDigest !== null && !digest(d.releaseDigest)) || typeof d.recoveryRequired !== 'boolean' || !['pending', 'confirmed'].includes(d.state) || (d.state === 'pending' ? d.visibleAt !== null : !iso(d.visibleAt))) throw new Error('分發狀態無效；不推定成功。');
  return Object.freeze({ ...d });
 }
 function statusReceipt(value, t) {
  if (!fields(value, ['version', 'gameId', 'releaseId', 'pin', 'distribution']) || value.version !== 1 || value.gameId !== t.gameId || !uuid(value.releaseId)) throw new Error('上架狀態回執無效。');
  const receipt = publishReceipt(value.pin, t);
  const distribution = distributionStatus(value.distribution, t.gameId, value.releaseId);
  if (result?.kind === 'publish' && (['gameId', 'name', 'ordinal', 'revision', 'sourceDigest', 'publishedAt'].some(k => result.receipt[k] !== receipt[k]) || JSON.stringify(result.receipt.chapters) !== JSON.stringify(receipt.chapters))) throw new Error('原上架回執不一致。');
  return { receipt, distribution };
 }
 // B8 全章匯總校驗：正確性只由服務端決定，本檔只做展示；回執逐項驗證，
 // 不接受未知形狀、超長文字或空問題列。
 function chapterIssues(value) {
  if (!Array.isArray(value) || value.length > 10000) throw new Error('全章校驗回應格式無效。');
  return value.map(chapter => {
   if (!fields(chapter, ['chapterId', 'chapterName', 'problems']) || typeof chapter.chapterId !== 'string' || !chapter.chapterId.length || chapter.chapterId.length > 256 || (chapter.chapterName !== null && (typeof chapter.chapterName !== 'string' || !chapter.chapterName.isWellFormed() || chapter.chapterName.length > 256)) || !Array.isArray(chapter.problems) || !chapter.problems.length || chapter.problems.length > 64) throw new Error('全章校驗回應格式無效。');
   return Object.freeze({ chapterId: chapter.chapterId, chapterName: chapter.chapterName, problems: Object.freeze(chapter.problems.map(problem => {
    if (!fields(problem, ['code', 'detail']) || typeof problem.code !== 'string' || !/^[A-Z][A-Z_]{2,63}$/.test(problem.code) || typeof problem.detail !== 'string' || !problem.detail.isWellFormed() || problem.detail.length > 512) throw new Error('全章校驗回應格式無效。');
    return Object.freeze({ code: problem.code, detail: problem.detail });
   })) });
  });
 }
 function referenceReceipt(value) {
  if (!fields(value, ['version', 'gameId', 'draftRevision', 'sourceDigest', 'chapterCount', 'chapters']) || value.version !== 1 || !uuid(value.gameId) || !revision(value.draftRevision) || !digest(value.sourceDigest) || !Number.isSafeInteger(value.chapterCount) || value.chapterCount < 0) throw new Error('全章校驗回應格式無效。');
  return Object.freeze({ draftRevision: value.draftRevision, chapterCount: value.chapterCount, chapters: Object.freeze(chapterIssues(value.chapters)) });
 }
 // 行內顯示：章名（找不到章時退為 id）＋每個原因；不把回應文字當成資料來源。
 const referenceRows = () => (references?.chapters ?? []).map(chapter => ({ chapterId: chapter.chapterId, name: chapter.chapterName ?? chapter.chapterId, problems: chapter.problems.map(problem => problem.detail) }));
 function referenceText() {
  if (!references) return '';
  const head = references.draftRevision === null ? '本次上架被拒絕（整遊戲校驗未通過）：' : '整遊戲校驗（修訂 ' + references.draftRevision + '，共 ' + references.chapterCount + ' 章）：';
  if (!references.chapters.length) return head + '全部章節引用完整。';
  return head + '有 ' + references.chapters.length + ' 章引用不到基礎資料：' + references.chapters.map(chapter => (chapter.chapterName ?? chapter.chapterId) + '（' + chapter.problems.map(problem => problem.detail).join('；') + '）').join('／');
 }
 function render() {
  const visible = current({ id: identity?.user.id, version: identity?.user.version, generation }), disabled = !visible || isBusy() || sending;
  const blocked = Boolean(pending) || badJournal;
  const list = visible ? rows.map(row => ({

   state: stateLabels[row.state] + (row.restricted ? '／管理限制' : ''), stateKey: row.state, rejected: row.rejected, createdAt: row.createdAt, stateTime: row.stateTime ?? '—', rowRevision: row.rowRevision,
   publishDisabled: disabled || blocked, deleteDisabled: disabled || blocked,
   deleteHint: '刪除整個遊戲（含所有章節與已上架版本，含資料庫列），不是刪除單個版本，徹底清除且不可恢復',
   publishTitle: row.state === 'draft' ? '校驗整個遊戲；通過後版本號自動為 1.0 並提交審核' : '校驗整個遊戲；通過後版本號自動 +1 並提交審核',
  })) : [];
  let tracking = '沒有追蹤中的上架或刪除操作。';
  if (badJournal) tracking = '本機記下的原操作已損壞，無法核對；為了不重複建立，暫時不接受新的上架或刪除（可清除本機追蹤）。';
  // 用戶要求（2026-11）：提示要讓人看懂現在是什麼狀態。這裡不再貼原鍵與 gameId，
  // 只說「發生了什麼、你還可以做什麼」；機器識別仍留在日誌與回饋裡。
  else if (pending) {
    const what = pending.kind === 'publish' ? '上架「' + pending.name + '」' : '刪除「' + pending.confirmationName + '」';
    tracking = '上一次' + what + '還沒拿到確定結果（連線可能中斷了）。'
      + (pending.version !== identity?.user.version ? '帳戶已變更，不能重送，請重新登入後再確認。'
        : checked ? '系統剛才查過一次，還沒找到結果，這不代表已取消；稍後可再查詢。'
        : '可以按「查詢原操作」確認結果；系統不會自動重送，也不會另外建第二份。');
  }
  let text = '';
  if (result?.kind === 'publish') {
   text = '本地已提交上架：版本 ' + result.receipt.ordinal + '／修訂 ' + result.receipt.revision + '。';
   text += distribution?.state === 'confirmed' ? ' 此次分發曾於 ' + distribution.visibleAt + ' 確認；上架與否仍以管理員同意為準。' : ' 分發待確認；尚未經管理員同意。';
   if (distribution?.recoveryRequired) text += ' 分發處於恢復鎖，須由管理者核對；本頁不解除。';
  } else if (result?.kind === 'delete') text = result.purge ? '已徹底刪除（不可恢復）：資料庫已移除 ' + result.purge.tables + ' 張表內該遊戲的所有列，玩家端目錄不再列出它；分發帳本與 R2 位元組由既有退役清理回收，不刪玩家本地存檔。' + reclaimText(result.purge.reclamation) : '本地已建立不可逆刪除柵欄（' + result.receipt.createdAt + '）；物理刪除尚未完成' + (result.purgeError ? '：' + String(result.purgeError).replace(/。+$/, '') : '（原因未回報）') + '。不刪玩家本地存檔。';
  emit({ rows: list, operation: visible ? tracking : '', result: visible ? text : '', message: visible ? messageText : '', references: visible ? referenceRows() : [], referencesText: visible ? referenceText() : '', locked: disabled,
   refreshDisabled: !visible || isBusy(), checkDisabled: !visible || isBusy() || sending,
   queryDisabled: disabled || !pending || badJournal,
   phase: visible ? phase : '',
   retryDisabled: disabled || !pending || badJournal || !checked || Boolean(result) || pending.version !== identity?.user.version,
   forgetDisabled: disabled || (!pending && !badJournal) });
 }
 async function load(ctx) {
  const value = await api('/api/games/submissions');
  if (!current(ctx)) return;
  rows = rowList(value);
 }
 async function refresh() { const ctx = capture(); await load(ctx); references = null; message(''); render(); return '上架清單已更新。'; }
 // 唯讀：不上架、不建鍵、不寫收據；修訂與摘要都由服務端回報。
 async function check(gameId) {
  const ctx = capture(), row = rows.find(entry => entry.gameId === gameId);
  if (!row) throw new Error('上架清單已變，請重新整理。');
  const value = await api('/api/games/' + gameId + '/draft/chapter-references');
  if (!current(ctx)) return;
  if (value?.gameId !== gameId) throw new Error('全章校驗回應的遊戲不符。');
  references = referenceReceipt(value);
  message(references.chapters.length ? '校驗失敗！' : '校驗成功！全部 ' + references.chapterCount + ' 章引用完整，可以上架（仍須管理員同意）。');
  render();
  // 頁面上方的提示要說清楚是什麼完成了。
  return references.chapters.length ? '檢驗未通過，請看下方清單。' : '檢驗完成。';
 }
 async function query() {
  const ctx = capture(), t = pending;
  if (!t || badJournal) throw new Error('沒有可核對的原操作關聯。');
  const serial = ++querySerial; checked = false; render();
  const live = () => current(ctx) && pending === t && serial === querySerial;
  const path = '/api/games/' + t.gameId + (t.kind === 'publish' ? '/publish-operations/' : '/deletion-operations/') + encodeURIComponent(t.key);
  try {
   const value = await api(path);
   if (!live()) return;
   if (t.kind === 'publish') { const found = statusReceipt(value, t); result = { kind: 'publish', receipt: found.receipt }; distribution = found.distribution; message('已核對原上架；本地提交、分發確認與管理員同意是不同事實。'); }
   else {
    // 查詢只回柵欄收據。若原刪除回應遺失，物理刪除那一步從未執行：這裡不能用同一個原鍵
    // 補跑（柵欄已成立），而是用 -purge 後綴鍵冪等重放，讓「徹底刪除」不會停在柵欄。
    const receipt = deleteReceipt(value, t), known = purgeOutcomes.get(t.key);
    if (known) { result = { kind: 'delete', receipt, ...known }; }
    else {
      try {
        const purge = purgeReceipt(await api('/api/admin/games/' + t.gameId + '/purge', { confirmationName: t.confirmationName, expectedRowRevision: receipt.rowRevision }, { key: t.key + '-purge' }));
        purgeOutcomes.set(t.key, { purge }); result = { kind: 'delete', receipt, purge };
      } catch (error) {
        const purgeError = errors[error.code] ?? String((error && (error.code || error.message)) || error || '原因未回報');
        purgeOutcomes.set(t.key, { purgeError }); result = { kind: 'delete', receipt, purgeError };
      }
    }
    message('已核對原刪除柵欄；物理刪除的完成證明看下方回執。');
   }
   render();
  } catch (error) {
   // 徹底刪除會連同刪除紀錄一起清除：查不到不是取消或回滾的證明。
   if (live() && t.kind === 'delete' && error.code === 'DELETE_OPERATION_NOT_FOUND') { message('原刪除紀錄已隨物理刪除清除；不推定取消或回滾，也不另建第二份。'); render(); throw error; }
   if (live()) { if (!result && (error.code === 'PUBLISH_OPERATION_NOT_FOUND' || error.code === 'DELETE_OPERATION_NOT_FOUND')) checked = true; message((errors[error.code] ?? error.message) + ' 保留同一個請求；不推定成功、失敗或取消。'); render(); }
   throw error;
  }
 }
 async function send(t, ctx) {
  if (sending || !current(ctx) || pending !== t || badJournal || t.version !== ctx.version) throw new Error('原操作關聯或帳戶已變。');
  sending = true; checked = false; querySerial++; phase = '正在提交' + (t.kind === 'publish' ? '上架' : '刪除') + '…'; render();
  try {
   const value = t.kind === 'publish'
    ? await api('/api/publish', { gameId: t.gameId, name: t.name }, { key: t.key })
    : await api('/api/games/' + t.gameId + '/deletion', { gameId: t.gameId, expectedRowRevision: t.expectedRowRevision, confirmationName: t.confirmationName }, { key: t.key });
   if (!current(ctx) || pending !== t) return;
   if (t.kind === 'publish') { result = { kind: 'publish', receipt: publishReceipt(value, t) }; }
   else {
    const fence = deleteReceipt(value, t); result = { kind: 'delete', receipt: fence };
    // 用戶決定（2026-11）：刪除＝徹底清除。柵欄成立後立刻物理刪除資料庫列；
    // 物理刪除失敗只降級成「已柵欄、尚未徹底刪除」，不推翻已成立的事實。
    try { const purge = purgeReceipt(await api('/api/admin/games/' + t.gameId + '/purge', { confirmationName: t.confirmationName, expectedRowRevision: fence.rowRevision }, { key: t.key + '-purge' })); purgeOutcomes.set(t.key, { purge }); result = { kind: 'delete', receipt: fence, purge }; }
    catch (error) { result = { kind: 'delete', receipt: fence, purgeError: errors[error.code] ?? String((error && (error.code || error.message)) || error || '原因未回報') }; }
   }
   distribution = null; phase = '已提交，正在讀取回執與清單…'; message('本地已提交，正在重新讀取清單。'); render();
   // A 路線第一版：伺服器已在本次請求內推進分發時，回執自帶描述子（與 GET 狀態同構）。
   if (t.kind === 'publish' && value.distribution !== undefined) distribution = distributionStatus(value.distribution, t.gameId, value.distribution.operationId);
   // A failed reload or observation never turns a committed local operation into failure.
   try { await load(ctx); } catch { /* list stays stale until the next explicit refresh */ }
   // 已確認就不再多走一次狀態查詢；待確認仍照舊查詢。
   if (t.kind === 'publish' && distribution?.state !== 'confirmed') { try { await query(); } catch { /* query keeps the original operation and reports uncertainty */ } }
   // 回執已證明本次請求的結果，關聯不再有待核對的事實：釋放追蹤鍵，讓清單可以再上架下一版。
   if (t.kind === 'publish') references = null;
   if (current(ctx) && pending === t) { try { store(null); } catch { /* 收據已成立；保留關聯只影響下一筆，不影響本次結果 */ } }
  } catch (error) {
   // 4xx 是明確拒絕：服務已答覆本次未提交，沒有需要核對的原鍵，修正後可直接重試。
   if (t.kind === 'publish' && error.code === 'PUBLISH_CHAPTER_REFERENCES' && current(ctx)) { try { references = Object.freeze({ draftRevision: null, chapterCount: null, chapters: Object.freeze(chapterIssues(error.detail?.chapters)) }); } catch { references = null; } }
   if (current(ctx) && pending === t) { if (Number.isSafeInteger(error.status) && error.status >= 400 && error.status < 500) { try { store(null); } catch { /* 保留原鍵只影響下一筆，不影響本次結論 */ } } message(errors[error.code] ?? error.message); render(); }
   throw error;
  } finally { sending = false; phase = ''; render(); }
 }
 async function submit(gameId) {
  const ctx = capture(), row = rows.find(entry => entry.gameId === gameId);
  if (!row) throw new Error('上架清單已變，請重新整理。');
  if (pending || badJournal) throw new Error('請先核對原操作，不另起鍵。');
  // 產品決定（2026-11）：上架前必須先校驗通過；未通過就不送鍵、不寫任何東西。
  phase = '正在校驗全部章節…'; render();
  await check(gameId);
  if (!current(ctx)) return;
  if (!references || references.chapters.length) { message('校驗未成功，無法上架。請先修正下方列出的章節問題。'); render(); return; }
  const t = Object.freeze({ actorId: ctx.id, kind: 'publish', version: ctx.version, gameId: row.gameId, key: crypto.randomUUID(), name: row.name });
  if (!tuple(t)) throw new Error(errors.PUBLISH_NAME);
  if (!current(ctx) || pending || badJournal || !rows.includes(row)) throw new Error('上架期間帳戶或清單已變；請重新整理。');
  store(t); result = null; distribution = null; render(); await send(t, ctx);
 }
 async function remove(gameId, rowRevision) {
  const ctx = capture(), row = rows.find(entry => entry.gameId === gameId && entry.rowRevision === rowRevision);
  if (!row) throw new Error('刪除目標已變，請重新整理清單。');
  if (pending || badJournal) throw new Error('請先核對原操作，不另起鍵。');
  const t = Object.freeze({ actorId: ctx.id, kind: 'delete', version: ctx.version, gameId: row.gameId, key: crypto.randomUUID(), expectedRowRevision: row.rowRevision, confirmationName: row.name });
  if (!tuple(t)) throw new Error('刪除確認名稱或請求格式無效。');
  if (!current(ctx) || pending || badJournal || !rows.includes(row)) throw new Error('刪除期間帳戶或清單已變；請重新整理。');
  store(t); result = null; distribution = null; render(); await send(t, ctx);
 }
 async function retry() {
  const ctx = capture(), t = pending, mark = querySerial;
  if (!t || !checked || result || badJournal || t.version !== ctx.version) throw new Error('須先查詢，且只能明確同鍵重送。');
  if (!confirm('以同一個請求與原內容重送？若已提交，只取原結果；若尚未提交，服務將校驗目前的已保存草稿。')) return;
  if (!current(ctx) || pending !== t || querySerial !== mark || badJournal || !checked || result) throw new Error('重試確認已過期，請重新查詢原操作。');
  await send(t, ctx);
 }
 function forget() {
  const ctx = capture(), original = pending, broken = badJournal;
  if (!confirm('只停止本分頁追蹤？不取消在途請求、不回滾或刪除服務收據，也不表示未生效。')) return;
  if (!current(ctx) || pending !== original || badJournal !== broken) return;
  store(null); result = null; distribution = null; message('已停止本機追蹤；服務原請求不受影響。'); render();
 }
 const action = fn => (...args) => work(async () => { const ctx = capture(); try { return await fn(...args); } catch (error) { if (current(ctx)) message((errors[error.code] ?? error.message) + (pending || badJournal ? ' 同一個請求保留；先查詢，不自動重送。' : ' 本次未提交；修正後可重新上架。')); throw error; } finally { render(); } });
 const actions = Object.freeze({ refresh: action(refresh), check: action(check), publish: action(submit), remove: action(remove), query: action(query), retry: action(retry), forget: action(forget) });
 return Object.freeze({ actions, render,
  setIdentity(value) { const next = value?.user ? { user: { id: value.user.id, version: value.user.version, role: value.user.role, mustChangePassword: value.user.mustChangePassword } } : null; const changed = JSON.stringify(identity) !== JSON.stringify(next); identity = next; if (changed) { generation++; rows = []; message(''); references = null; restore(); } render(); },
  // 上架面是整遊戲清單，沒有自己的「當前遊戲」；它跟隨共享選擇（gameContext），
  // 這樣五個面板對「現在處理哪個遊戲」永遠同一個答案。
  selectedGame: () => (allowed() ? gameContext?.current() ?? null : null),
  canAdoptGame(id) { return uuid(id) && current({ id: identity?.user.id, version: identity?.user.version, generation }); },
  adoptGame(id) { if (!uuid(id)) return; render(); },
  canLeave(forUnload = false) { return (!pending && !badJournal) || (!forUnload && confirm('原上架或刪除操作仍保留在本分頁。離開不取消請求；返回後須明確查詢。')); } });
}
