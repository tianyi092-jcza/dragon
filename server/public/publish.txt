 // 上架/刪除被暫停時要能直接說出原因（用戶 m1389：按鈕灰掉卻不知道為什麼）。

 // 動作失敗時接在原因後面的一句話：先說「現在該做什麼」，並且要能認出追蹤的是哪一款。
 function stuckHint() {

 }
// 上架申請＝送出一次審核申請（管理員同意後玩家端才可見）；按鈕只在「有東西可送」且「本次修訂
// 已通過服務端檢驗」时開放：草稿（從未發佈）→ 需先檢驗；待審核中→ 不可重複送；已上架→ 只有
// 草稿有新修訂才需要送下一版；被駁回→ 可修正後重送。

 function validationLabel(row) {
  if (!row.validation) return '未檢驗';
  if (!fresh(row)) return '檢驗已過期（草稿已更新）';
  return row.validation.state === 'passed' ? '檢驗成功' : '檢驗失敗（' + row.validation.failedChapters + ' 章）';
 }
 function validationHint(row) {
  if (!row.validation) return '尚未檢驗：先按「檢驗」跑完整遊戲校驗，通過後才能送出上架申請';
  if (!fresh(row)) return '上次檢驗是修訂 ' + row.validation.draftRevision + '，目前草稿已是修訂 ' + row.draftRevision + '；請重新檢驗';
  if (row.validation.state === 'failed') return '上次檢驗未通過（' + row.validation.failedChapters + ' 章引用不到基礎資料）；修正後重新檢驗';
  return '';
 }
 function submittable(row) {
  if (!validated(row)) return false;
  if (row.state === 'draft') return true;
  if (row.state === 'pending') return row.rejected;
  return row.draftChanged;
 }
 function submitHint(row) {
  const next = '校驗整個遊戲；通過後版本號自動為 ' + nextVersion(row.version) + ' 並提交管理員審核';
  if (row.state === 'draft') return next;
  if (row.state === 'pending') return row.rejected ? '上一次上架申請已被駁回；修正後可重新校驗並送出（版本 ' + nextVersion(row.version) + '）' : '上架申請已送出，等管理員在「審核」按「同意」；同一版本不能重複提交';
  return row.draftChanged ? next : '已上架且沒有新修訂；先修改資料並保存新修訂，才需要上架下一版';
 }

 async function load(ctx) {
  const value = await api('/api/games/submissions');
  if (!current(ctx)) return false;
  rows = rowList(value); listed = true; return reconcile();
 }
 // 用戶要求（2026-11）：刪掉那個遊戲就等於那筆追蹤失去對象，不該再擋住後來新遊戲的上架。
 // 判據只認「清單已成功讀取、且該 gameId 不在其中」——讀取失敗時 rows 保留舊值，不會誤清；
 // 本分頁自己正在送的那一筆（sending）不在此列，它的結論由 send() 決定。
 function reconcile() {
  if (!pending || !listed || sending) return false;
  if (rows.some(row => row.gameId === pending.gameId)) return false;
  const gone = targetText(pending);
  try { store(null); } catch { return false; }
  result = null; distribution = null; checked = false;
  message('上一次' + gone + '：目標已刪除，本機追蹤已自動作廢；可以直接上架目前清單中的遊戲。');
  return true;
 }
 async function refresh() { const ctx = capture(); const voided = await load(ctx); references = null; if (!voided) message(''); render(); return '上架清單已更新。'; }
 // 唯讀：不上架、不建鍵、不寫收據；修訂與摘要都由服務端回報。
 async function check(gameId) {
  const ctx = capture(), row = rows.find(entry => entry.gameId === gameId);
  if (!row) throw new Error('上架清單已變，請重新整理。');
  // 檢驗是服務端動作（POST）：逐層各問一次，算完一層才更新一層；四層到齊才落庫。
  // 因此遮罩裡的計時與四層進度是同時發生的，不是等全部算完才一次顯示。
  checkedLayers = new Set(); references = null; render();
  const merged = new Map();
  let head = null;
  for (const layer of layers) {
   const value = await api('/api/games/' + gameId + '/draft/chapter-references', { layer: layer.key });
   if (!current(ctx)) return;
   if (value?.gameId !== gameId) throw new Error('全章校驗回應的遊戲不符。');
   const receipt = referenceReceipt(value);
   head ??= { draftRevision: receipt.draftRevision, chapterCount: receipt.chapterCount };
   if (head.draftRevision !== receipt.draftRevision || head.chapterCount !== receipt.chapterCount) throw new Error('全章校驗回應的修訂不一致。');
   for (const chapter of receipt.chapters) {
    const bucket = merged.get(chapter.chapterId) ?? { chapterId: chapter.chapterId, chapterName: chapter.chapterName, problems: [] };
    bucket.problems.push(...chapter.problems);
    merged.set(chapter.chapterId, bucket);
   }
   references = Object.freeze({ draftRevision: head.draftRevision, chapterCount: head.chapterCount, chapters: Object.freeze([...merged.values()]) });
   checkedLayers.add(layer.key);
   render();
  }
  // 結果已由服務端保存，重讀清單才能用同一份事實決定按鈕可用性。
  try { await load(ctx); } catch { /* 清單留待手動刷新；本次檢驗結果仍在下方 */ }
  if (!current(ctx)) return;
  // 按鈕是否真的可送要用同一份清單事實判斷：已上架且沒有新修訂時要說「先存新修訂」，
  // 不能一邊說「可以送出上架申請」一邊把按鈕鎖著。
  const after = rows.find(entry => entry.gameId === gameId), paused = blockedText() || (after && !submittable(after) ? submitHint(after) : '');
  message(references.chapters.length ? '檢驗失敗！' : '檢驗成功！全部 ' + references.chapterCount + ' 章引用完整' + (paused ? '；' + paused : '，可以送出上架申請（仍須管理員同意）。'));
  render();
  // 頁面上方的提示要說清楚是什麼完成了。
  return references.chapters.length ? '檢驗未通過，請看下方清單。' : '檢驗完成。';
 }
 async function query() {
  const ctx = capture(), t = pending;
  if (!t || badJournal) throw new Error('沒有可核對的原操作關聯。');
  const serial = ++querySerial; checked = false; render();
  const live = () => current(ctx) && pending === t && serial === querySerial;
  const path = '/api/games/' + t.gameId + (t.kind === 'publish' ? '/publish-operations/' : '/deletion-operations/') + encodeURIComponent(t.key);
  try {
   const value = await api(path);
   if (!live()) return;
   if (t.kind === 'publish') {
    const found = statusReceipt(value, t); result = { kind: 'publish', receipt: found.receipt }; distribution = found.distribution;
    message('已核對原上架申請：' + targetText(t) + '；本地提交、分發確認與管理員同意是不同事實。');
    // 核對到回執就代表清單已變（例如草稿→待上架）：順手重讀，避免畫面仍顯示舊狀態。
    try { await load(ctx); } catch { /* 清單留待下次手動刷新 */ }
    if (!live()) return;
    // 回執已證明這筆請求的結果，追蹤鍵不再有待核對的事實：與上架成功同樣結案，
    // 不然會停在「查到了、按鈕還是灰的」那一步。
    try { store(null); } catch { /* 收據已成立；保留追蹤只影響下一筆 */ }
   }
   else {
    // 查詢只回柵欄收據。若原刪除回應遺失，物理刪除那一步從未執行：這裡不能用同一個原鍵
    // 補跑（柵欄已成立），而是用 -purge 後綴鍵冪等重放，讓「徹底刪除」不會停在柵欄。
    const receipt = deleteReceipt(value, t), known = purgeOutcomes.get(t.key);
    if (known) { result = { kind: 'delete', receipt, ...known }; }
    else {
      try {
        const purge = purgeReceipt(await api('/api/admin/games/' + t.gameId + '/purge', { confirmationName: t.confirmationName, expectedRowRevision: receipt.rowRevision }, { key: t.key + '-purge' }));
        purgeOutcomes.set(t.key, { purge }); result = { kind: 'delete', receipt, purge };
      } catch (error) {
        const purgeError = errors[error.code] ?? String((error && (error.code || error.message)) || error || '原因未回報');
        purgeOutcomes.set(t.key, { purgeError }); result = { kind: 'delete', receipt, purgeError };
      }
    }
    message('已核對原刪除柵欄；物理刪除的完成證明看下方回執。');
   }
   render();
  } catch (error) {
   // 伺服器答「這個 game 不存在」＝權威證明追蹤對象已消失，本機記錄再無可核對的事實：直接作廢。
   if (live() && error.code === 'GAME_NOT_FOUND') {
    try { store(null); } catch { /* 記錄留著只影響下一筆 */ }
    result = null; distribution = null; checked = false;
    message('上一次' + targetText(t) + '：目標已刪除，本機追蹤已自動作廢；可以直接上架目前清單中的遊戲。');
    render(); return;
   }
   // 徹底刪除會連同刪除紀錄一起清除：查不到不是取消或回滾的證明。
   if (live() && t.kind === 'delete' && error.code === 'DELETE_OPERATION_NOT_FOUND') { message('原刪除紀錄已隨物理刪除清除；不推定取消或回滾，也不另建第二份。'); render(); throw error; }
   if (live()) { if (!result && (error.code === 'PUBLISH_OPERATION_NOT_FOUND' || error.code === 'DELETE_OPERATION_NOT_FOUND')) checked = true; message((errors[error.code] ?? error.message) + ' 保留同一個請求；不推定成功、失敗或取消。'); render(); }
   throw error;
  }
 }
 // 一筆上架只送一次 POST；結果由「先到的確定答案」決定：POST 回執，或同一把鍵的冪等查詢回執。
 // 只有連線層失敗（沒有 HTTP 狀態碼）才算「未知」而轉入輪詢；服務端任何答覆（含 4xx／5xx）都照原樣拋出。
 async function publishOutcome(t, ctx) {
  const path = '/api/games/' + t.gameId + '/publish-operations/' + encodeURIComponent(t.key), started = Date.now();
  startProgress(path, t, ctx);
  let lost = false;
  const post = api('/api/publish', { gameId: t.gameId, name: t.name }, { key: t.key }).then(
   value => ({ kind: 'post', value }),
   error => { if (Number.isSafeInteger(error?.status)) { throw error; } lost = true; return { kind: 'lost' }; });
  // 輪詢先行拿到結果後，POST 的成敗不再影響結論；但也不留下未處理的拒絕。
  post.catch(() => {});
  while (current(ctx) && pending === t) {
   if (!lost) {
    const answer = await waitFor(post, PUBLISH_POLL_MS);
    if (answer && answer.kind === 'post') return answer;
   } else { await wait(PUBLISH_POLL_MS); }
   // POST 還沒有答案：問一下伺服器這把鍵是否已經落地。
   try { return { kind: 'status', status: statusReceipt(await api(path), t) }; }
   catch (error) {
    // 「還沒有紀錄」＝還在編譯；5xx／斷線＝暫時問不到。兩者都繼續等。
    // 其餘 4xx 是明確拒絕（例如權限已變），照原樣拋出，不能拖到超時。
    const status = error?.status, code = error?.code;
    if (Number.isSafeInteger(status) && status >= 400 && status < 500 && code !== 'PUBLISH_OPERATION_NOT_FOUND') throw error;
   }
   if (Date.now() - started > PUBLISH_POLL_BUDGET_MS) return { kind: 'unknown' };
   phase = '伺服器仍在處理上架（已 ' + Math.round((Date.now() - started) / 1000) + ' 秒；拿到結果會自動關閉）…';
   render();
  }
  return { kind: 'unknown' };
 }
 async function send(t, ctx) {
  if (sending || !current(ctx) || pending !== t || badJournal || t.version !== ctx.version) throw new Error('原操作關聯或帳戶已變。');
  sending = true; checked = false; querySerial++; phase = '正在提交' + (t.kind === 'publish' ? '上架' : '刪除') + '…'; render();
  try {
   const outcome = t.kind === 'publish'
    ? await publishOutcome(t, ctx)
    : { kind: 'post', value: await api('/api/games/' + t.gameId + '/deletion', { gameId: t.gameId, expectedRowRevision: t.expectedRowRevision, confirmationName: t.confirmationName }, { key: t.key }) };
   if (!current(ctx) || pending !== t) return;
   if (outcome.kind === 'unknown') {
    phase = '';
    message('上架請求已送出，但 ' + (PUBLISH_POLL_BUDGET_MS / 60000) + ' 分鐘內沒取得確定結果；伺服器可能仍在處理，請稍後按「查詢原操作」核對，不要重送。');
    render(); return;
   }
   const value = outcome.value;
   if (t.kind === 'publish') { result = { kind: 'publish', receipt: outcome.kind === 'status' ? outcome.status.receipt : publishReceipt(value, t) }; }
   else {
    const fence = deleteReceipt(value, t); result = { kind: 'delete', receipt: fence };
    // 用戶決定（2026-11）：刪除＝徹底清除。柵欄成立後立刻物理刪除資料庫列；
    // 物理刪除失敗只降級成「已柵欄、尚未徹底刪除」，不推翻已成立的事實。
    try { const purge = purgeReceipt(await api('/api/admin/games/' + t.gameId + '/purge', { confirmationName: t.confirmationName, expectedRowRevision: fence.rowRevision }, { key: t.key + '-purge' })); purgeOutcomes.set(t.key, { purge }); result = { kind: 'delete', receipt: fence, purge }; }
    catch (error) { result = { kind: 'delete', receipt: fence, purgeError: errors[error.code] ?? String((error && (error.code || error.message)) || error || '原因未回報') }; }
   }
   // 輪詢得來的狀態本身就是回執＋分發描述子；POST 回執則看它有沒有附分發。
   distribution = outcome.kind === 'status' ? outcome.status.distribution : null;
   phase = '已提交，正在讀取回執與清單…'; message('本地已提交，正在重新讀取清單。'); render();
   // A 路線第一版：伺服器已在本次請求內推進分發時，回執自帶描述子（與 GET 狀態同構）。
   if (t.kind === 'publish' && outcome.kind === 'post' && value.distribution !== undefined) distribution = distributionStatus(value.distribution, t.gameId, value.distribution.operationId);
   // A failed reload or observation never turns a committed local operation into failure.
   try { await load(ctx); } catch { /* list stays stale until the next explicit refresh */ }
   // 已確認就不再多走一次狀態查詢；待確認仍照舊查詢。
   if (t.kind === 'publish' && outcome.kind === 'post' && distribution?.state !== 'confirmed') { try { await query(); } catch { /* query keeps the original operation and reports uncertainty */ } }
   // 回執已證明本次請求的結果，關聯不再有待核對的事實：釋放追蹤鍵，讓清單可以再上架下一版。
   if (t.kind === 'publish') references = null;
   if (current(ctx) && pending === t) { try { store(null); } catch { /* 收據已成立；保留關聯只影響下一筆，不影響本次結果 */ } }
  } catch (error) {
   // 4xx 是明確拒絕：服務已答覆本次未提交，沒有需要核對的原鍵，修正後可直接重試。
   if (t.kind === 'publish' && error.code === 'PUBLISH_CHAPTER_REFERENCES' && current(ctx)) { try { references = Object.freeze({ draftRevision: null, chapterCount: null, chapters: Object.freeze(chapterIssues(error.detail?.chapters)) }); } catch { references = null; } }
   if (current(ctx) && pending === t) { if (Number.isSafeInteger(error.status) && error.status >= 400 && error.status < 500) { try { store(null); } catch { /* 保留原鍵只影響下一筆，不影響本次結論 */ } } message(errors[error.code] ?? error.message); render(); }
   throw error;
  } finally { sending = false; phase = ''; stopProgress(); render(); }
 }
 async function submit(gameId) {
  const ctx = capture();
  let row = rows.find(entry => entry.gameId === gameId);
  if (!row) throw new Error('上架清單已變，請重新整理。');
  if (!submittable(row)) throw new Error(validationHint(row) || submitHint(row));
  if (pending || badJournal) throw new Error('請先核對原操作，不另起鍵。');
  // 清單已有本次修訂的通過記錄就直接送出；沒有才先跑一次服務端檢驗（POST 落庫）。
  if (!validated(row)) {
   phase = '正在檢驗全部章節…'; render();
   await check(gameId);
   if (!current(ctx)) return;
   if (!references || references.chapters.length) { message('檢驗未成功，無法送出上架申請。請先修正下方列出的章節問題。'); render(); return; }
   row = rows.find(entry => entry.gameId === gameId) ?? row;
  }
  const t = Object.freeze({ actorId: ctx.id, kind: 'publish', version: ctx.version, gameId: row.gameId, key: crypto.randomUUID(), name: row.name });
  if (!tuple(t)) throw new Error(errors.PUBLISH_NAME);
  if (!current(ctx) || pending || badJournal || !rows.includes(row)) throw new Error('上架申請期間帳戶或清單已變；請重新整理。');
  store(t); result = null; distribution = null; render(); await send(t, ctx);
 }
 async function remove(gameId, rowRevision) {
  const ctx = capture(), row = rows.find(entry => entry.gameId === gameId && entry.rowRevision === rowRevision);
  if (!row) throw new Error('刪除目標已變，請重新整理清單。');
  if (pending || badJournal) throw new Error('請先核對原操作，不另起鍵。');
  const t = Object.freeze({ actorId: ctx.id, kind: 'delete', version: ctx.version, gameId: row.gameId, key: crypto.randomUUID(), expectedRowRevision: row.rowRevision, confirmationName: row.name });
  if (!tuple(t)) throw new Error('刪除確認名稱或請求格式無效。');
  if (!current(ctx) || pending || badJournal || !rows.includes(row)) throw new Error('刪除期間帳戶或清單已變；請重新整理。');
  store(t); result = null; distribution = null; render(); await send(t, ctx);
 }
 async function retry() {
  const ctx = capture(), t = pending, mark = querySerial;
  if (!t || !checked || result || badJournal || t.version !== ctx.version) throw new Error('須先查詢，且只能明確同鍵重送。');
  if (!confirm('以同一個請求與原內容重送？若已提交，只取原結果；若尚未提交，服務將校驗目前的已保存草稿。')) return;
  if (!current(ctx) || pending !== t || querySerial !== mark || badJournal || !checked || result) throw new Error('重試確認已過期，請重新查詢原操作。');
  await send(t, ctx);
 }
 function forget() {
  const ctx = capture(), original = pending, broken = badJournal;
  if (!confirm('只停止本分頁追蹤？不取消在途請求、不回滾或刪除服務收據，也不表示未生效。')) return;
  if (!current(ctx) || pending !== original || badJournal !== broken) return;
  store(null); result = null; distribution = null; message('已停止本機追蹤；服務原請求不受影響。'); render();
 }
 const action = fn => (...args) => work(async () => { const ctx = capture(); try { return await fn(...args); } catch (error) { if (current(ctx)) message((errors[error.code] ?? error.message) + stuckHint()); throw error; } finally { render(); } });
 const actions = Object.freeze({ refresh: action(refresh), check: action(check), publish: action(submit), remove: action(remove), query: action(query), retry: action(retry), forget: action(forget) });
 return Object.freeze({ actions, render,
  setIdentity(value) { const next = value?.user ? { user: { id: value.user.id, version: value.user.version, role: value.user.role, mustChangePassword: value.user.mustChangePassword } } : null; const changed = JSON.stringify(identity) !== JSON.stringify(next); identity = next; if (changed) { generation++; rows = []; listed = false; message(''); references = null; restore(); } render(); },
  // 上架面是整遊戲清單，沒有自己的「當前遊戲」；它跟隨共享選擇（gameContext），
  // 這樣五個面板對「現在處理哪個遊戲」永遠同一個答案。
  selectedGame: () => (allowed() ? gameContext?.current() ?? null : null),
  canAdoptGame(id) { return uuid(id) && current({ id: identity?.user.id, version: identity?.user.version, generation }); },
  adoptGame(id) { if (!uuid(id)) return; render(); },
  canLeave(forUnload = false) { return (!pending && !badJournal) || (!forUnload && confirm('原上架申請或刪除操作仍保留在本分頁。離開不取消請求；返回後須明確查詢。')); } });
}
