// Read-only reference adapter. Numeric editing belongs exclusively to the
// shared draft journal; this adapter never posts or acknowledges a save.
export function installStronghold({ api, authority, work, isBusy, publish = () => {}, gameContext }) {
  let identity = null, generation = 0, actionGeneration = 0, games = [], selected = null, detail = null, summary = null, chapterId = null, message = '';
  const allowed = () => Boolean(identity?.user) && identity === authority() && !identity.user.mustChangePassword;
  const chapter = () => (detail?.chapters ?? []).find(c => c.chapterId === chapterId) ?? null;
  const city = idx => chapter()?.cities.find(c => c.idx === idx);
  const refsFor = idx => detail?.strongholds?.find(r => r.runtimeSlot === idx);
  const canLeave = () => true;
  function clear() { games = []; selected = gameContext?.current() ?? null; detail = null; summary = null; chapterId = null; message = ''; }
  function adoptGame(value) { if (!allowed() || value === selected) return; generation++; actionGeneration++; selected = value; detail = null; summary = null; chapterId = null; message = ''; render(); }
  function render() {
    const permitted = allowed(), locked = !permitted || isBusy();
    publish({ selected: permitted ? selected : null, chapterId: permitted ? chapterId : null, locked, saveDisabled: true,
      games: permitted ? games.map(g => ({gameId:g.gameId,label:g.metadata.name + '／草稿 ' + g.draftRevision})) : [],
      chapters: permitted ? (detail?.chapters ?? []).map(c => ({chapterId:c.chapterId,name:c.name || c.chapterId})) : [],
      cities: permitted ? (chapter()?.cities ?? []).map(c => { const refs = refsFor(c.idx); return {idx:c.idx,cityId:refs?.cityId ?? null,name:c.name,faction:c.faction ?? '—',roadCount:refs?.roads?.length ?? 0,chapterCount:refs?.chapters?.length ?? 0,troopsCap:c.troopsCap,maxProd:c.maxProd,troops:String(c.troops ?? ''),prod:String(c.prod ?? '')}; }) : [],
      context: permitted && summary ? summary.gameId + '／已讀取修訂 ' + (detail?.draftRevision ?? summary.draftRevision) : '', message:permitted ? message : '', dirty:false });
  }
  const revision = value => typeof value === 'string' && /^[1-9][0-9]{0,63}$/.test(value);
  function acceptGames(data) {
    if (!Array.isArray(data?.games)) throw new Error('遊戲清單回應無效。');
    const ids = new Set();
    return data.games.map(g => { if (typeof g?.gameId !== 'string' || !g.gameId || ids.has(g.gameId) || !revision(g.draftRevision) || typeof g.metadata?.name !== 'string') throw new Error('遊戲清單回應無效。'); ids.add(g.gameId); return {gameId:g.gameId,draftRevision:g.draftRevision,metadata:{name:g.metadata.name}}; });
  }
  function acceptDetail(data, target) {
    if (data?.gameId !== target.gameId || data.draftRevision !== target.draftRevision || !data.metadata || typeof data.metadata.name !== 'string' || !Array.isArray(data.chapters) || !Array.isArray(data.strongholds)) throw new Error('據點回應身份／修訂無效。');
    const ids = new Set();
    for (const c of data.chapters) {
      if (typeof c?.chapterId !== 'string' || ids.has(c.chapterId) || !Array.isArray(c.cities)) throw new Error('章節據點清單無效。'); ids.add(c.chapterId);
      const slots = new Set();
      for (const row of c.cities) { if (!Number.isSafeInteger(row?.idx) || slots.has(row.idx) || typeof row.name !== 'string') throw new Error('章節據點清單無效。'); slots.add(row.idx); }
    }
    const definitionIds = new Set(), definitionSlots = new Set();
    for (const ref of data.strongholds) {
      if (!Number.isSafeInteger(ref?.runtimeSlot) || ref.runtimeSlot < 0 || ref.runtimeSlot >= 192 || definitionSlots.has(ref.runtimeSlot)) throw new Error('據點定義槽身份無效。');
      definitionSlots.add(ref.runtimeSlot);
      if (ref.cityId !== undefined && ref.cityId !== null) {
        if (typeof ref.cityId !== 'string' || !ref.cityId || definitionIds.has(ref.cityId)) throw new Error('據點定義ID身份無效。');
        definitionIds.add(ref.cityId);
      }
      if (ref?.references !== undefined && (!Array.isArray(ref.references) || ref.references.some(row => typeof row?.chapterId !== 'string' || typeof row.code !== 'string' || typeof row.detail !== 'string'))) throw new Error('據點引用報告無效。');
    }
    return structuredClone(data);
  }
  function ticket() { const actor = identity, serial = ++generation; return () => allowed() && identity === actor && serial === generation; }
  async function inspect() {
    if (!allowed()) throw new Error('請先登入。');
    const target = games.find(g => g.gameId === selected); if (!target) throw new Error('請先選擇遊戲。');
    const captured = {...target}, valid = ticket();
    const result = await api('/api/games/' + encodeURIComponent(captured.gameId) + '/draft?revision=' + encodeURIComponent(captured.draftRevision));
    if (!valid()) return false;
    const next = acceptDetail(result, captured); if (!valid()) return false;
    detail = next; summary = captured;
    if (!detail.chapters.some(c => c.chapterId === chapterId)) chapterId = detail.chapters[0]?.chapterId ?? null;
    message = '此表只讀取據點引用；數值編輯請使用共用草稿數值表。基礎 192 鎖定不開放增刪。';
    return true;
  }
  async function refresh() {
    if (!allowed()) throw new Error('請先登入。');
    const valid = ticket(); if (!canLeave() || !valid()) return;
    const result = await api('/api/games'); if (!valid()) return;
    games = acceptGames(result);
    const next = gameContext?.current() ?? (games.some(game => game.gameId === selected) ? selected : games[0]?.gameId ?? null);
    if (next && gameContext && !gameContext.select(next)) return;
    if (!valid()) return;
    if (next !== selected) { selected = next; chapterId = null; }
    detail = null; summary = null;
    if (games.some(game => game.gameId === selected)) await inspect();
    else message = '目前遊戲不在本次清單；沒有切換其他遊戲。';
  }
  function action(callback) { return work(async () => { const actor = identity, serial = ++actionGeneration; try { await callback(); } catch (error) { if (identity === actor && allowed() && serial === actionGeneration) message = error instanceof Error ? error.message : String(error); throw error; } finally { render(); } }); }
  const actions = Object.freeze({
    refresh: () => action(refresh), inspect: () => action(async () => { const valid = ticket(); if (canLeave() && valid()) await inspect(); }),
    selectGame(value) { if (!allowed() || isBusy() || value === selected) return; const valid = ticket(); if (!canLeave() || !valid()) { render(); return; } if (!games.some(g => g.gameId === value)) return; if (gameContext && !gameContext.select(value)) { render(); return; } if (!valid()) return; adoptGame(value); },
    selectChapter(value) { if (!allowed() || isBusy() || value === chapterId) return; const valid = ticket(); if (!canLeave() || !valid()) { render(); return; } if (!detail?.chapters.some(c => c.chapterId === value)) return; generation++; chapterId = value; render(); },
    blockers: idx => action(async () => { if (!allowed()) return; const c = city(idx); if (!c) return; const r = refsFor(idx), list = [];
      if (!r) list.push('查無引用資料，先讀取管理清單。');
      else { if (r.placed) list.push('已放入地圖，須先從地圖移除'); for (const road of r.roads ?? []) list.push(`道路引用：${road}`); for (const ch of r.chapters ?? []) list.push(`章節引用：${ch}`); for (const ref of r.references ?? []) list.push(`${ref.chapterId}：${ref.detail}`); if (r.references === undefined) list.push('詳細角色引用未提供，不能視為無引用'); list.push('僅列已讀章節字段，不代表事件／原始資料完整刪除閉包'); list.push('基礎據點鎖定 192 座，不開放增刪'); }
      message = `「${c.name}」不可刪除：${list.join('；')}。`;
    }),
  });
  render(); return Object.freeze({render, actions, canLeave, adoptGame, canAdoptGame: () => allowed(), selectedGame: () => allowed() ? selected : null, setIdentity(value) { if (identity !== value) { generation++; actionGeneration++; identity = value; clear(); } render(); }});
}
