// Map viewer (§4.4 B4d1, read-only schematic): placements + road geometry
// from draft read. No editing: placements are 192-locked with frozen cells,
// road topology is chapter-pinned, tile semantics unaudited. This view is the
// visual foundation for future editing, not an editor.
export function installMapViewer({ api, authority, work, isBusy, publish = () => {}, gameContext, getCanvas = () => null }) {
  let identity = null, games = [], selected = null, detail = null, topo = null, generation = 0, actionGeneration = 0, message = '';
  const allowed = () => Boolean(identity?.user) && identity === authority() && !identity.user.mustChangePassword;
  function render() {
    const visible = allowed(), lines = [];
    if (visible && detail?.map) {
      const roads = detail.map.roads ?? [], moved = new Set((topo?.moves ?? []).map(m => m.cityId));
      lines.push(`放置 ${detail.map.placements ?? 0}／道路 ${roads.length}／邊界 384×256`);
      if (topo) lines.push(`拓撲意向 ${(topo.moves ?? []).length}／斷路 ${roads.filter(r => moved.has(r.fromCityId) || moved.has(r.toCityId)).length}／已登記重建 ${(topo.rebuilds ?? []).length}／${topo.blocked ? '存盘发布阻挡' : '無阻挡'}`);
      else lines.push('拓撲意向未確認；不代表無阻擋或可以發布。');
      for (const road of roads.slice(0,200)) lines.push(`${road.id ?? '?'}：${road.fromCityId} → ${road.toCityId}（${(road.geometry ?? []).length} 點，${road.travelKind ?? '?'}）`);
      if (roads.length > 200) lines.push(`…另 ${roads.length - 200} 條未列出`);
    }
    publish({ locked: !visible || isBusy(), selected: visible ? selected : null,
      games: visible ? games.map(g => ({gameId:g.gameId,label:g.metadata.name + '／草稿 ' + g.draftRevision})) : [],
      lines, message: visible ? message : '', context: visible && detail ? detail.gameId + '／草稿修訂 ' + detail.draftRevision : '' });
    const canvas = getCanvas(), context = canvas?.getContext('2d');
    if (!context) return;
    context.clearRect(0, 0, canvas.width, canvas.height);
    if (!visible || !detail?.map) return;
    const roads = detail.map.roads ?? [];
    // Receipt carries road geometry; dots use authoritative base coords.
    const dots = detail.strongholds ?? [];
    const bounds = { minX: 0, minY: 0, width: 384, height: 256 };
    const sx = canvas.width / bounds.width, sy = canvas.height / bounds.height;
    context.fillStyle = '#222';
    context.fillRect(0, 0, canvas.width, canvas.height);
    context.strokeStyle = '#4a6b3a';
    context.lineWidth = 1;
    for (const road of roads) {
      const pts = road.geometry ?? [];
      if (pts.length < 2) continue;
      context.beginPath();
      context.moveTo((pts[0].x - bounds.minX) * sx, (pts[0].y - bounds.minY) * sy);
      for (const pt of pts.slice(1)) context.lineTo((pt.x - bounds.minX) * sx, (pt.y - bounds.minY) * sy);
      context.stroke();
    }
    context.fillStyle = '#e8d9b0';
    for (const dot of dots) {
      context.fillRect((dot.x - bounds.minX) * sx - 1, (dot.y - bounds.minY) * sy - 1, 3, 3);
    }
    // B-3b topology overlay (display only): intended positions (amber),
    // broken roads (red), rebuilt-recorded count as text. Derived locally
    // from the read receipt + intent list; never mutates drafts.
    const moved = new Map((topo?.moves ?? []).map(m => [m.cityId, m]));
    if (moved.size) {
      const byId = new Map(dots.map(d => [d.cityId, d]));
      context.strokeStyle = '#d08000';
      context.fillStyle = '#d08000';
      for (const [cityId, m] of moved) {
        const dot = byId.get(cityId);
        if (dot) {
          context.beginPath();
          context.moveTo((dot.x - bounds.minX) * sx, (dot.y - bounds.minY) * sy);
          context.lineTo((m.x - bounds.minX) * sx, (m.y - bounds.minY) * sy);
          context.stroke();
        }
        context.fillRect((m.x - bounds.minX) * sx - 2, (m.y - bounds.minY) * sy - 2, 5, 5);
      }
      context.strokeStyle = '#d00000';
      for (const road of roads) {
        if (!moved.has(road.fromCityId) && !moved.has(road.toCityId)) continue;
        const pts = road.geometry ?? [];
        if (pts.length < 2) continue;
        context.beginPath();
        context.moveTo((pts[0].x - bounds.minX) * sx, (pts[0].y - bounds.minY) * sy);
        for (const pt of pts.slice(1)) context.lineTo((pt.x - bounds.minX) * sx, (pt.y - bounds.minY) * sy);
        context.stroke();
      }
    }
  }
  function adoptGame(value) { if (!allowed() || value === selected) return; generation++; actionGeneration++; selected = value; detail = null; topo = null; message = ''; render(); }
  function ticket() { const actor = identity, serial = ++generation; return () => allowed() && actor === identity && serial === generation; }
  async function refresh() {
    if (!allowed()) throw new Error('請先登入。');
    const valid = ticket(), result = await api('/api/games'); if (!valid()) return;
    const ids = new Set();
    if (!Array.isArray(result.games) || result.games.some(g => !g || typeof g.gameId !== 'string' || ids.has(g.gameId) || !ids.add(g.gameId) || typeof g.metadata?.name !== 'string' || !/^[1-9][0-9]*$/.test(g.draftRevision))) throw new Error('遊戲清單回應無效。');
    games = result.games.map(g => ({gameId:g.gameId,draftRevision:String(g.draftRevision),metadata:{name:g.metadata.name}}));
    const next = gameContext?.current() ?? (games.some(game => game.gameId === selected) ? selected : games[0]?.gameId ?? null);
    if (next && gameContext && !gameContext.select(next)) return;
    if (!valid()) return;
    selected = next;
    detail = null; topo = null; message = ''; render();
    if (games.some(game => game.gameId === selected)) await inspect();
    else message = '目前遊戲不在本次清單；沒有切換其他遊戲。';
  }
  // This schematic supports only the current fixed rectangle, not map admission.
  const point = value => Number.isInteger(value?.x) && value.x >= 0 && value.x < 384 && Number.isInteger(value?.y) && value.y >= 0 && value.y < 256;
  function uniqueRows(rows, key, check) {
    if (!Array.isArray(rows)) return false;
    const ids = new Set();
    for (const row of rows) {
      const id = row?.[key];
      if (typeof id !== 'string' || !id || ids.has(id) || !check(row)) return false;
      ids.add(id);
    }
    return true;
  }
  async function inspect() {
    if (!allowed()) throw new Error('請先登入。');
    const game = games.find(entry => entry.gameId === selected); if (!game) throw new Error('請先選擇遊戲。');
    const valid = ticket(), next = await api('/api/games/' + encodeURIComponent(game.gameId) + '/draft?revision=' + encodeURIComponent(game.draftRevision));
    if (!valid()) return;
    if (next?.gameId !== game.gameId || next.draftRevision !== game.draftRevision || !Array.isArray(next.strongholds) || !Array.isArray(next.map?.roads)) throw new Error('地圖回應身份／修訂無效。');
    if (!uniqueRows(next.strongholds, 'cityId', point) || !uniqueRows(next.map.roads, 'id', road => Array.isArray(road.geometry) && Array.from(road.geometry).every(point))) throw new Error('地圖幾何回應無效；保留先前已讀示意圖。');
    const cities = new Set(next.strongholds.map(city => city.cityId)), roads = new Set(next.map.roads.map(road => road.id));
    let overlay = null;
    try { overlay = await api('/api/games/' + encodeURIComponent(game.gameId) + '/topology'); }
    catch { /* Optional read failure is unknown, never an unblocked result. */ }
    if (!valid()) return;
    if (overlay && (overlay.gameId !== game.gameId || !uniqueRows(overlay.moves, 'cityId', move => point(move) && cities.has(move.cityId)) || !uniqueRows(overlay.rebuilds, 'roadId', rebuild => roads.has(rebuild.roadId) && ['land', 'water'].includes(rebuild.travelKind)) || overlay.blocked !== (overlay.moves.length > 0))) overlay = null;
    detail = structuredClone(next); topo = structuredClone(overlay);
    message = '示意圖唯讀：放置／道路／圖塊編輯尚未開放（模型鎖定，見 §4.4 缺口）。'; render();
  }
  const action = callback => work(async () => { const actor = identity, serial = ++actionGeneration; try { await callback(); } catch (error) { if (allowed() && identity === actor && serial === actionGeneration) message = error.message; throw error; } finally { render(); } });
  const actions = Object.freeze({redraw:render,refresh:()=>action(refresh),inspect:()=>action(inspect),select(value) { if (!allowed() || isBusy() || !games.some(g => g.gameId === value) || value === selected) return; const valid = ticket(); if (gameContext && !gameContext.select(value)) { render(); return; } if (!valid()) return; adoptGame(value); }});
  render(); return Object.freeze({ render, actions, adoptGame, canAdoptGame: () => allowed(), selectedGame: () => allowed() ? selected : null, setIdentity(value) { if (identity !== value) { generation++; actionGeneration++; identity = value; games = []; selected = gameContext?.current() ?? null; detail = null; topo = null; message = ''; } render(); }, canLeave() { return true; } });
}
